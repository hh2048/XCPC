## 第八章：数据结构

约定：
- 数组下标、元素排名等均从 $1$ 开始（第 $0$ 项/行/列默认作为补零哨兵）；
- 查询区间为闭区间 $[l, r]$；
- 相同名称的封装需独立选择，不能同时复制。

模板非百科全书，如存在多个数据结构可以维护同一信息，默认仅保留复杂度最优的版本。

### 前缀和与差分：离线区间修改、静态区间查询

空间与初始化复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(1)$。常见操作为区间和、区间异或和。

```c++
vector<i64> d(n + 2);
i64 last = 0;
for (int i = 1; i <= n; i++) { // 由初始数组 a 构造差分
    d[i] = a[i] - last;
    last = a[i];
}
auto rangeAdd = [&](int l, int r, i64 v) {
    d[l] += v;
    d[r + 1] -= v;
};

// 所有修改完成后还原 a
for (int i = 1; i <= n; i++) {
    d[i] += d[i - 1];
    a[i] = d[i];
}

vector<i64> pre(n + 1);
for (int i = 1; i <= n; i++) {
    pre[i] = pre[i - 1] + a[i];
}
auto rangeSum = [&](int l, int r) {
    return pre[r] - pre[l - 1];
};
```

### 二维前缀和与差分：离线矩形修改、静态矩形查询

空间与初始化复杂度 $\mathcal O(nm)$，单次操作复杂度 $\mathcal O(1)$。

```c++
vector<vector<i64>> d(n + 2, vector<i64>(m + 2));
for (int x = 1; x <= n; x++) {
    for (int y = 1; y <= m; y++) { // 由初始矩阵 a 构造二维差分
        d[x][y] = a[x][y] - a[x - 1][y] - a[x][y - 1] + a[x - 1][y - 1];
    }
}
auto matrixAdd = [&](int u, int v, int x, int y, i64 p) { // [u, v] ~ [x, y] 全部加 p
    x++, y++;
    d[u][v] += p, d[x][v] -= p;
    d[x][y] += p, d[u][y] -= p;
};

// 所有修改完成后还原 a
for (int x = 1; x <= n; x++) {
    for (int y = 1; y <= m; y++) {
        d[x][y] += d[x - 1][y] + d[x][y - 1] - d[x - 1][y - 1];
        a[x][y] = d[x][y];
    }
}

vector<vector<i64>> pre(n + 1, vector<i64>(m + 1));
for (int x = 1; x <= n; x++) {
    for (int y = 1; y <= m; y++) {
        pre[x][y] = a[x][y] + pre[x - 1][y] + pre[x][y - 1] - pre[x - 1][y - 1];
    }
}
auto matrixSum = [&](int u, int v, int x, int y) {
    return pre[x][y] - pre[u - 1][y] - pre[x][v - 1] + pre[u - 1][v - 1];
};
```

### ST 表：静态区间查询

用于静态数组的幂等区间查询（$x \text{ 运算符 } x=x$），**不支持修改操作**。空间与初始化复杂度 $\mathcal O(n \log n)$，单次操作复杂度 $\mathcal O(1)$。

`Op` 为无状态、可默认构造的函数对象，按左区间、右区间的顺序合并，必须满足结合律与幂等律。常见操作为区间最小值（$\min(x,x)=x$）、最大值（$\max(x,x)=x$）、$\gcd$（$\gcd(x,x)=x$）、按位与、按位或。

```c++
template<class Op> struct ST {
    int n;
    vector<vector<i64>> st;

    ST(const vector<i64> &a) : n(int(a.size()) - 1) {
        st.push_back(a);
        for (int k = 1; (1 << k) <= n; k++) {
            int len = 1 << k, half = len >> 1;
            st.emplace_back(n - len + 2);
            for (int l = 1; l + len - 1 <= n; l++) {
                st[k][l] = Op{}(st[k - 1][l], st[k - 1][l + half]);
            }
        }
    }
    i64 query(int l, int r) const { // 要求 1 <= l <= r <= n
        int k = __lg(r - l + 1);
        return Op{}(st[k][l], st[k][r - (1 << k) + 1]);
    }
};
```

### 树状数组：单点修改、区间查询

空间复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$。

#### 单点加、区间和

```c++
struct BIT {
    int n;
    vector<i64> w;
    BIT(int n) : n(n), w(n + 1) {}

    void pointAdd(int x, i64 v) { // 单点 x 修改
        for (; x <= n; x += x & -x) {
            w[x] += v;
        }
    }
    i64 pointSum(int x) { return rangeSum(x, x); } // 单点 x 查询
    i64 rangeSum(int l, int r) { // 区间和 [l, r] 查询
        auto sum = [&](int x) {
            i64 ans = 0;
            for (; x; x -= x & -x) {
                ans += w[x];
            }
            return ans;
        };
        return sum(r) - sum(l - 1);
    }
};
```

例如，计算逆序对数（支持相等元素，注意数组的下标是从 $1$ 开始的）：

```c++
    i64 ans = 0;
    for (auto [v, x] : ord) {
```

#### 双树状数组：区间加、区间和

常数比上一个版本略微大一些。

```c++
struct RangeBIT {
    int n;
    vector<i64> a, b;
    RangeBIT(int n) : n(n), a(n + 1), b(n + 1) {}
    
    void pointAdd(int x, i64 v) { rangeAdd(x, x, v); } // 单点 x 修改
    void rangeAdd(int l, int r, i64 v) { // 区间 [l, r] 修改
        auto add = [&](int x, i64 v) {
            for (int i = x; i <= n; i += i & -i) {
                a[i] += v;
                b[i] += x * v;
            }
        };
        add(l, v), add(r + 1, -v);
    }
    i64 pointSum(int x) { return rangeSum(x, x); } // 单点 x 查询
    i64 rangeSum(int l, int r) { // 区间和 [l, r] 查询
        auto sum = [&](int x) {
            i64 ans = 0;
            for (int i = x; i; i -= i & -i) {
                ans += (x + 1) * a[i] - b[i];
            }
            return ans;
        };
        return sum(r) - sum(l - 1);
    }
};
```

#### 值域树状数组：集合排名、第 k 小、前后继

适用于值域有界或可离线离散化，且需要排名或第 $k$ 小；仅需插入、删除、前后继时，直接使用 `multiset`。

注意事项：
- 构造函数传入待维护值的闭区间 $[\text{lo}, \text{hi}]$，所有操作值必须落在其中；
- **离线时优先离散化**；值域过大且无法离线离散化时，使用平衡树替代；
- 删除元素时，需要自行保证目标元素存在；
- 查找前驱与后继时，应当自行保证严格前驱/后继存在；
- 查找第 $k$ 小时，需要自行保证第 $k$ 小存在。

`add` 函数在这里用于插入或删除元素：`bit.add(x, 1)`、`bit.add(x, -1)`。

```c++
struct RankBIT {
    int n; i64 lo;
    vector<int> w;
    RankBIT(i64 lo, i64 hi) : n(int(hi - lo + 1)), lo(lo), w(n + 1) {}

    int id(i64 x) const { return int(x - lo + 1); } // 偏移
    void add(i64 x, int v) { // 单点 x 修改
        for (int p = id(x); p <= n; p += p & -p) {
            w[p] += v;
        }
    }
    int sum(int p) const {
        int ans = 0;
        for (; p; p -= p & -p) {
            ans += w[p];
        }
        return ans;
    }

    i64 kth(int k) const { // 查找第 k 小的值
        int ans = 0;
        for (int i = 1 << __lg(n); i; i >>= 1) {
            int next = ans + i;
            if (next <= n && w[next] < k) {
                k -= w[next];
                ans = next;
            }
        }
        return lo + ans;
    }
    int rank(i64 x) const { return sum(id(x) - 1) + 1; } // 查找 x 的排名
    i64 pre(i64 x) const { return kth(rank(x) - 1); } // 查找 x 的前驱
    i64 suf(i64 x) const { return kth(sum(id(x)) + 1); } // 查找 x 的后继
};
```

### 并查集

除可撤销并查集外，均支持路径压缩和按大小合并，均摊复杂度为 $\mathcal O(\alpha(n))$（比 $\log n$ 更小，在 $n = 2 \times 10^5$ 时，$\alpha(n) < 5$）。根编号不固定。

#### 常用封装

```c++
struct DSU {
    vector<int> fa, siz;
    int cnt; // 当前连通块数量

    DSU(int n) : fa(n + 1), siz(n + 1, 1), cnt(n) {
        iota(fa.begin(), fa.end(), 0);
    }
    int get(int x) {
        while (x != fa[x]) {
            x = fa[x] = fa[fa[x]];
        }
        return x;
    }
    bool merge(int x, int y) { // 返回是否合并两个连通块
        x = get(x), y = get(y);
        if (x == y) return false;
        if (siz[x] < siz[y]) swap(x, y);
        fa[y] = x;
        siz[x] += siz[y];
        cnt--;
        return true;
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    int size(int x) { // 返回 x 所在的连通块中点的数量
        return siz[get(x)];
    }
};
```

#### 扩展封装（维护无向图连通块信息）

仅用于无向图，可额外维护无向图连通块数量、点数和边数（重边与自环正常计数）。

```c++
struct DSU {
    vector<int> fa, siz, edges;
    int cnt; // 连通块数量

    DSU(int n) : fa(n + 1), siz(n + 1, 1), edges(n + 1), cnt(n) {
        iota(fa.begin(), fa.end(), 0);
    }
    int get(int x) {
        while (x != fa[x]) {
            x = fa[x] = fa[fa[x]];
        }
        return x;
    }
    bool add(int x, int y) { // 加入一条无向边，返回是否合并两个连通块
        x = get(x), y = get(y);
        if (x == y) {
            edges[x]++;
            return false;
        }
        if (siz[x] < siz[y]) swap(x, y);
        fa[y] = x;
        siz[x] += siz[y];
        edges[x] += edges[y] + 1;
        cnt--;
        return true;
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    int size(int x) { // 返回 x 所在的连通块中点的数量
        return siz[get(x)];
    }
    int edgeCount(int x) { // 返回 x 所在的连通块中边的数量
        return edges[get(x)];
    }
    bool isTree(int x) { // 判断 x 所在的连通块是否为树
        x = get(x);
        return edges[x] == siz[x] - 1;
    }
};
```

#### 带权并查集

当题目逐条给出两点间的差值或关系，并要求判断新陈述能否与已有陈述同时成立时使用。构造时 `mod = 0` 表示普通整数差值；`mod = 2` 可处理奇偶 / 二分图关系，`mod = 3` 可处理食物链等循环关系。

```c++
struct wDSU {
    vector<int> fa, siz;
    vector<i64> dis; // val[x] - val[fa[x]]
    i64 mod;

    wDSU(int n, i64 M = 0) : fa(n + 1), siz(n + 1, 1), dis(n + 1), mod(M) {
        iota(fa.begin(), fa.end(), 0);
    }
    i64 norm(i64 x) const {
        if (!mod) return x;
        x %= mod;
        return x < 0 ? x + mod : x;
    }
    int get(int x) {
        if (x == fa[x]) return x;
        int p = fa[x];
        fa[x] = get(p);
        dis[x] = norm(dis[x] + dis[p]);
        return fa[x];
    }
    bool add(int x, int y, i64 w) { // val[x] - val[y] = w，返回新陈述是否相容
        int rx = get(x), ry = get(y);
        w = norm(w);
        if (rx == ry) return norm(dis[x] - dis[y]) == w;
        i64 rootDiff = norm(w - dis[x] + dis[y]); // val[rx] - val[ry]
        if (siz[rx] < siz[ry]) {
            fa[rx] = ry;
            dis[rx] = rootDiff;
            siz[ry] += siz[rx];
        } else {
            fa[ry] = rx;
            dis[ry] = norm(-rootDiff);
            siz[rx] += siz[ry];
        }
        return true; // 注意，即使两点已经连通，只要关系正确仍返回 true
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    i64 diff(int x, int y) { // 要求 same(x, y)
        get(x), get(y);
        return norm(dis[x] - dis[y]);
    }
};
```

#### 可撤销并查集

不能路径压缩，仅成功合并会写入历史，`get`、`same`、`merge` 的复杂度为 $\mathcal O(\log n)$，一次 `revert(t)` 的复杂度与撤销的成功合并次数成正比。

```c++
struct rDSU {
    vector<int> fa, siz;
    vector<array<int, 2>> his;
    int cnt;

    rDSU(int n) : fa(n + 1), siz(n + 1, 1), cnt(n) {
        iota(fa.begin(), fa.end(), 0);
    }
    int get(int x) { // 不能路径压缩
        while (x != fa[x]) x = fa[x];
        return x;
    }
    bool merge(int x, int y) {
        x = get(x), y = get(y);
        if (x == y) return false;
        if (siz[x] < siz[y]) swap(x, y);
        his.push_back({x, y}); // 仅记录成功合并
        fa[y] = x;
        siz[x] += siz[y];
        cnt--;
        return true;
    }
    int time() const { // 返回当前快照数量
        return his.size();
    }
    void revert(int t) { // 恢复到时间戳 t 的快照
        while (time() > t) {
            auto [x, y] = his.back();
            his.pop_back();
            fa[y] = y;
            siz[x] -= siz[y];
            cnt++;
        }
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    int size(int x) {
        return siz[get(x)];
    }
};
```

<blockquote>

例如，《P3402. 【模板】可持久化并查集》构建一个离线可持久化并查集，每个操作以某个旧版本（`op[u]`）为基础，先将版本依赖建成树（`son[u]`）。

```c++
}
```

### 静态线段树：单点修改、静态区间查询

适用于区间信息能由左右子区间合并、修改只落在单个位置的题目。空间与初始化复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$。

使用 zkw 风格，自底向上迭代封装，常数更优。

#### 统一封装（Info）

需要定义两项内容：
- `Info` 存储一个区间的信息，且 `Info()` 表示空区间；
- `operator+` 按“左区间 + 右区间”合并，必须满足结合律，**不要求交换律**。

```c++
template<class Info> struct SegTree {
    int n = 0;
    vector<Info> t;

    SegTree(const vector<Info> &a) {
        n = int(a.size()) - 1;
        t.assign(2 * n, Info());
        for (int i = 1; i <= n; i++) {
            t[n + i - 1] = a[i];
        }
        for (int p = n - 1; p; p--) {
            pull(p);
        }
    }
    void pull(int p) { t[p] = t[p << 1] + t[p << 1 | 1]; }
    void pointSet(int x, const Info &v) { // 单点 x 赋值为 v
        t[x += n - 1] = v;
        for (x >>= 1; x; x >>= 1) {
            pull(x);
        }
    }
    Info rangeQuery(int l, int r) const { // 查询闭区间 [l, r]
        Info u, v;
        for (l += n - 1, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) u = u + t[l++]; // 若信息不满足交换律（矩阵乘法、字符串拼接），必须注意顺序
            if (r & 1) v = t[--r] + v; // 若信息不满足交换律（矩阵乘法、字符串拼接），必须注意顺序
        }
        return u + v;
    }
};
```

再例如区间最小/最大值查询。

维护多个查询参数时，推荐补全构造函数，使用起来会更方便一些。

```c++
    i64 mx = numeric_limits<i64>::lowest();

    Info() = default;
    Info(i64 x) : mn(x), mx(x) {} // 区间维护多参数时，这个构造函数允许使用简单的单值初始化叶子节点
```

#### 常见查询速查

| 查询内容 | 成员默认值，即 `Info()` | 左右合并 | 备注 |
| :---: | --- | :---: | :---: |
| 区间和 | `sum = 0` | 加法 | 直接使用树状数组更佳 |
| 区间最小值 | `mn = numeric_limits<i64>::max()` | `min` |
| 区间最大值 | `mx = numeric_limits<i64>::lowest()` | `max` |
| 区间 gcd | `g = 0` | `gcd` |
| 区间异或 | `xr = 0` | 按位异或 |
| 区间按位或 | `mask = 0` | 按位或 |
| 区间按位与 | `mask = ~i64(0)` | 按位与 |

`gcd(0, x) = |x|`，因此 $0$ 是 gcd 的单位元。

最大值的单位元必须使用 `lowest()`；`min()` 并不表示最小可表示值。

#### 区间最大子段和

维护非空最大子段和。`sum` 为区间和，`lmx`、`rmx` 分别为最大前缀和、最大后缀和，`mx` 为最大子段和。

```c++
struct Info {
    static constexpr i64 P = -(1LL << 60); // 必须小于所有可能的非空子段和
    i64 sum = 0;
    i64 lmx = P, rmx = P, mx = P;

    Info() = default;
    Info(i64 x) : sum(x), lmx(x), rmx(x), mx(x) {}
    Info(i64 s, i64 l, i64 r, i64 m) : sum(s), lmx(l), rmx(r), mx(m) {}
};
Info operator+(const Info &a, const Info &b) {
    return {
        a.sum + b.sum,
        max(a.lmx, a.sum + b.lmx),
        max(b.rmx, b.sum + a.rmx),
        max({a.mx, b.mx, a.rmx + b.lmx})
    };
}

vector<Info> a = {{}, {-2}, {3}, {-1}, {4}, {-5}};
SegTree<Info> t(a);

t.pointSet(1, {5});
```

### 懒标记线段树：区间修改、区间查询

适用于区间修改能在 $\mathcal O(1)$ 内直接更新整个区间信息的题目。整体复杂度与不带懒标记的线段树封装相同，空间与初始化复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$，但常数更大。

使用标准递归风格，常数略大，但是兼容性更好。

#### 统一封装（Info + Tag）

需要额外定义：
- `Tag::apply(v)`：将新操作 `v` 叠加到已有标记之后，下传时等价于先执行旧标记，再执行 `v`。
- `Info::apply(v, len)`：将标记 `v` 作用到长度为 `len` 的区间信息。

封装内部用 `has` 判断标记是否存在，`Tag` 无需用特殊值表示空操作。

```c++
template<class Info, class Tag> struct LazySegTree {
    int n;
    vector<Info> info;
    vector<Tag> tag;
    vector<char> has; // 用于判断 tag[p] 是否有效

    LazySegTree(const vector<Info> &a) {
        n = int(a.size()) - 1;
        info.assign(4 * n + 4, Info());
        tag.resize(4 * n + 4);
        has.assign(4 * n + 4, 0);
        build(1, 1, n, a);
    }
    void pointSet(int x, const Info &v) { // 单点 x 赋值为 v
        pointSet(1, 1, n, x, v);
    }
    void rangeApply(int l, int r, const Tag &v) { // 区间 [l, r] 修改
        rangeApply(1, 1, n, l, r, v);
    }
    Info rangeQuery(int l, int r) { // 查询区间 [l, r]
        return rangeQuery(1, 1, n, l, r);
    }

private:
    void build(int p, int l, int r, const vector<Info> &a) {
        if (l == r) return info[p] = a[l], void();
        int m = (l + r) / 2;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        pull(p);
    }
    void pull(int p) {
        info[p] = info[p << 1] + info[p << 1 | 1];
    }
    void apply(int p, int l, int r, const Tag &v) {
        info[p].apply(v, r - l + 1);
        if (has[p]) tag[p].apply(v);
        else tag[p] = v, has[p] = 1;
    }
    void push(int p, int l, int r) {
        if (!has[p] || l == r) return;
        int m = (l + r) / 2;
        apply(p << 1, l, m, tag[p]);
        apply(p << 1 | 1, m + 1, r, tag[p]);
        has[p] = 0;
    }
    void pointSet(int p, int l, int r, int x, const Info &v) {
        if (l == r) return info[p] = v, has[p] = 0, void();
        push(p, l, r);
        int m = (l + r) / 2;
        if (x <= m) pointSet(p << 1, l, m, x, v);
        else pointSet(p << 1 | 1, m + 1, r, x, v);
        pull(p);
    }
    void rangeApply(int p, int l, int r, int x, int y, const Tag &v) {
        if (x <= l && r <= y) return apply(p, l, r, v);
        push(p, l, r);
        int m = (l + r) / 2;
        if (x <= m) rangeApply(p << 1, l, m, x, y, v);
        if (m < y) rangeApply(p << 1 | 1, m + 1, r, x, y, v);
        pull(p);
    }
    Info rangeQuery(int p, int l, int r, int x, int y) {
        if (x <= l && r <= y) return info[p];
        push(p, l, r);
        int m = (l + r) / 2;
        if (y <= m) return rangeQuery(p << 1, l, m, x, y);
        if (m < x) return rangeQuery(p << 1 | 1, m + 1, r, x, y);
        return rangeQuery(p << 1, l, m, x, y) + rangeQuery(p << 1 | 1, m + 1, r, x, y);
    }
};
```

仅维护区间加与区间和时，使用树状数组更佳。

#### 区间赋值、区间加

`Tag::f` 表示该操作中是否包含赋值。

这里同时维护区间和、区间最小值、区间最大值查询。维护多个查询参数时，推荐补全构造函数，使用起来会更方便一些。

```c++
struct Tag {
    bool f = false;
    i64 set = 0, add = 0;
    void apply(const Tag &v) { // 先执行本标记，再执行 v
        if (v.f) {
            f = true;
            set = v.set;
            add = v.add;
        } else {
            add += v.add;
        }
    }
};
struct Info {
    i64 sum = 0;
    i64 mn = numeric_limits<i64>::max();
    i64 mx = numeric_limits<i64>::lowest();

    Info() = default;
    Info(i64 x) : sum(x), mn(x), mx(x) {}
    Info(i64 s, i64 lo, i64 hi) : sum(s), mn(lo), mx(hi) {}

        if (v.f) {
            sum = v.set * len;
            mn = mx = v.set;
        }
        sum += v.add * len;
        mn += v.add;
        mx += v.add;
```

维护区间和查询时，需要额外维护 `B` 个计数器，统计每一位为 $1$ 的元素个数。设所有初值和异或值满足 $0 \le x < 2^B$。

空间与初始化复杂度 $\mathcal O(Bn)$，单次操作复杂度 $\mathcal O(B\log n)$。

```c++
constexpr int B = 31; // 保证所有数都在 [0, 2^B) 内

struct Tag {
    i64 mask = 0;
    void apply(const Tag &v) { mask ^= v.mask; }
};
struct Info {
    array<int, B> cnt{};
    
    Info() = default;
    Info(i64 x) {
        for (int i = 0; i < B; i++) {
            cnt[i] = x >> i & 1;
        }
    }
    void apply(const Tag &v, int len) {
        for (int i = 0; i < B; i++) {
            if (v.mask >> i & 1) {
                cnt[i] = len - cnt[i];
            }
        }
    }
    i64 sum() const {
        i64 res = 0;
        for (int i = 0; i < B; i++) {
            res += 1LL * cnt[i] * (1LL << i);
        }
        return res;
    }
};
Info operator+(const Info &a, const Info &b) {
    Info c;
    for (int i = 0; i < B; i++) {
        c.cnt[i] = a.cnt[i] + b.cnt[i];
    }
    return c;
}

vector<Info> a = {{}, {1}, {2}, {3}, {4}, {5}};
LazySegTree<Info, Tag> t(a);

t.rangeApply(2, 4, {3}); // [1, 1, 0, 7, 5]

cout << t.rangeQuery(1, 5).sum() << "\n"; // 14
```

### 线段树思想：特殊修改与结构变换

包含一些相对高频、但无法简单使用统一封装的特殊修改操作。这些操作无法在 $\mathcal O(1)$ 内由整段信息完成修改，因此不能直接套用上一节的统一封装（Info + Tag）。

虽然理论上依旧可以使用势能线段树做统一封装，但是实现起来会比较复杂，故在此单独列出。

#### 区间取模/取整开方、单点赋值、区间和

本质是势能剪枝，无法仅由区间和直接计算取模后的区间和。额外维护区间最大值：
- 取模时，若区间最大值小于模数，整段不变；两次赋值之间，单点真正取模至多 $\mathcal O(\log(A+1))$ 次。
- 开方时，若区间最大值不超过 $1$，整段不变；两次赋值之间，单点真正开方至多 $\mathcal O(\log\log(A+2))$ 次。

其中，$A$ 为初值及所有赋入值的最大值。要求所有元素始终非负，且每次模数 $v > 0$。

总均摊复杂度为 $\mathcal O((n+q)\log n\log(A+2))$。

```c++
template<class T> T isqrt(T n) { // n >= 0
    T x = sqrtl(n); // 如不需要兼容 `__int128`，可使用 `sqrt` 代替，常数略小一点
    while (x + 1 <= n / (x + 1)) ++x;
    while (x && x > n / x) --x;
    return x;
}

struct SegTreeShrink {
    int n;
    vector<i64> sum, mx;

    SegTreeShrink(const vector<i64> &a) : n(int(a.size()) - 1), sum(4 * n + 4), mx(4 * n + 4) {
        build(1, 1, n, a);
    }

    void pointSet(int x, i64 v) { pointSet(1, 1, n, x, v); }
    void rangeMod(int l, int r, i64 v) { rangeMod(1, 1, n, l, r, v); }
    void rangeSqrt(int l, int r) { rangeSqrt(1, 1, n, l, r); }
    i64 rangeSum(int l, int r) { return rangeSum(1, 1, n, l, r); }

private:
    void build(int p, int l, int r, const vector<i64> &a) {
        if (l == r) return sum[p] = mx[p] = a[l], void();
        int m = (l + r) / 2;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        pull(p);
    }
    void pull(int p) {
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
        mx[p] = max(mx[p << 1], mx[p << 1 | 1]);
    }
    void pointSet(int p, int l, int r, int x, i64 v) {
        if (l == r) return sum[p] = mx[p] = v, void();
        int m = (l + r) / 2;
        if (x <= m) pointSet(p << 1, l, m, x, v);
        else pointSet(p << 1 | 1, m + 1, r, x, v);
        pull(p);
    }
    void rangeMod(int p, int l, int r, int x, int y, i64 v) {
        if (mx[p] < v) return; // 整段取模后不变
        if (l == r) return sum[p] %= v, mx[p] = sum[p], void();
        int m = (l + r) / 2;
        if (x <= m) rangeMod(p << 1, l, m, x, y, v);
        if (m < y) rangeMod(p << 1 | 1, m + 1, r, x, y, v);
        pull(p);
    }
    void rangeSqrt(int p, int l, int r, int x, int y) {
        if (mx[p] <= 1) return; // 0、1 开方后不变
        if (l == r) return sum[p] = mx[p] = isqrt(sum[p]), void();
        int m = (l + r) / 2;
        if (x <= m) rangeSqrt(p << 1, l, m, x, y);
        if (m < y) rangeSqrt(p << 1 | 1, m + 1, r, x, y);
        pull(p);
    }
    i64 rangeSum(int p, int l, int r, int x, int y) {
        if (x <= l && r <= y) return sum[p];
        int m = (l + r) / 2;
        i64 ans = 0;
        if (x <= m) ans += rangeSum(p << 1, l, m, x, y);
        if (m < y) ans += rangeSum(p << 1 | 1, m + 1, r, x, y);
        return ans;
    }
};
```

#### 区间加、单点赋值、区间 gcd 查询

本质是维护差分前缀和，令 $d_1=a_1$，$d_i=a_i-a_{i-1}$。区间加 $l,r$ 加上 $v$ 时，只会改变两个位置：$d_l \to d_l+v$、$d_{r+1} \to d_{r+1}-v \left(r<n \right)$。由定义：

$$
\gcd(a_l,a_{l+1},\ldots,a_r)
=\gcd(a_l,d_{l+1},d_{l+2},\ldots,d_r)
$$

故树状数组维护差分前缀和，得到 $a_l$；线段树维护差分绝对值的 $\gcd$。

空间与初始化复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$。

```c++
struct RangeAddGcd {
    int n;
    vector<i64> d, bit, seg;

    RangeAddGcd(const vector<i64> &a) : n(int(a.size()) - 1), d(n + 1), bit(n + 1), seg(2 * n) {
        for (int i = 1; i <= n; i++) {
            d[i] = a[i] - (i == 1 ? 0 : a[i - 1]);
            bitAdd(i, d[i]);
            seg[n + i - 1] = gcd(d[i], 0LL);
        }
        for (int p = n - 1; p; p--) {
            seg[p] = gcd(seg[p << 1], seg[p << 1 | 1]);
        }
    }

    void rangeAdd(int l, int r, i64 v) { // a[l..r] += v
        changeD(l, v);
        changeD(r + 1, -v);
    }
    void pointSet(int x, i64 v) { // a[x] = v
        rangeAdd(x, x, v - pointValue(x));
    }
    i64 pointValue(int x) const { return prefixSum(x); }
    i64 rangeGcd(int l, int r) const {
        return gcd(pointValue(l), rangeGcdDiff(l + 1, r));
    }

private:
    void bitAdd(int x, i64 v) {
        for (; x <= n; x += x & -x) bit[x] += v;
    }
    i64 prefixSum(int x) const {
        i64 ans = 0;
        for (; x; x -= x & -x) ans += bit[x];
        return ans;
    }
    void changeD(int x, i64 v) { // d[x] += v
        if (x > n) return;
        d[x] += v;
        bitAdd(x, v);

        int p = n + x - 1;
        seg[p] = gcd(d[x], 0LL);
        for (p >>= 1; p; p >>= 1) {
            seg[p] = gcd(seg[p << 1], seg[p << 1 | 1]);
        }
    }
    i64 rangeGcdDiff(int l, int r) const {
        if (l > r) return 0;
        i64 ans = 0;
        for (l += n - 1, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) ans = gcd(ans, seg[l++]);
            if (r & 1) ans = gcd(ans, seg[--r]);
        }
        return ans;
    }
};
```

### 二维树状数组：矩形修改、矩形查询

空间与初始化复杂度均为 $\mathcal O(nm)$，单次操作复杂度 $\mathcal O(\log n\log m)$。

#### 单点加、矩形和

```c++
struct BIT_2D {
    int n, m;
    vector<vector<i64>> w;

    BIT_2D(int n, int m) : n(n), m(m), w(n + 1, vector<i64>(m + 1)) {}
    void pointAdd(int x, int y, i64 p) { // 单点 [x, y] 修改
        for (int i = x; i <= n; i += i & -i) {
            for (int j = y; j <= m; j += j & -j) {
                w[i][j] += p;
            }
        }
    }
    i64 pointSum(int x, int y) { return matrixSum(x, y, x, y); } // 单点 [x, y] 查询
    i64 matrixSum(int u, int v, int x, int y) { // 矩形和 [u, v] ~ [x, y] 查询
        auto sum = [&](int x, int y) {
            i64 ans = 0;
            for (int i = x; i; i -= i & -i) {
                for (int j = y; j; j -= j & -j) {
                    ans += w[i][j];
                }
            }
            return ans;
        };
        return sum(x, y) - sum(u - 1, y) - sum(x, v - 1) + sum(u - 1, v - 1);
    }
};
```

#### 矩形加、单点和

```c++
struct BIT_2D {
    int n, m;
    vector<vector<i64>> w;

    BIT_2D(int n, int m) : n(n), m(m), w(n + 1, vector<i64>(m + 1)) {}
    void pointAdd(int x, int y, i64 p) { matrixAdd(x, y, x, y, p); } // 单点 [x, y] 修改
    void matrixAdd(int u, int v, int x, int y, i64 p) { // 矩形 [u, v] ~ [x, y] 修改
        auto add = [&](int x, int y, i64 p) {

                for (int j = y; j <= m; j += j & -j) {
                    w[i][j] += p;
                }
            }

        add(u, v, p), add(x, v, -p);
        add(x, y, p), add(u, y, -p);
    }

        i64 ans = 0;
        for (int i = x; i; i -= i & -i) {
            for (int j = y; j; j -= j & -j) {
                ans += w[i][j];
            }
        }
        return ans;
    }
};
```

#### 矩形加、矩形和

常数比上两个版本略微大一些。

```c++
struct exBIT_2D {
    int n, m;
    vector<vector<i64>> b1, b2, b3, b4;

    exBIT_2D(int n, int m) : n(n), m(m),
        b1(n + 1, vector<i64>(m + 1)), b2(n + 1, vector<i64>(m + 1)),
        b3(n + 1, vector<i64>(m + 1)), b4(n + 1, vector<i64>(m + 1)) {}
    void pointAdd(int x, int y, i64 p) { matrixAdd(x, y, x, y, p); } // 单点 [x, y] 修改
    void matrixAdd(int u, int v, int x, int y, i64 p) { // 矩形 [u, v] ~ [x, y] 修改
        auto add = [&](int u, int v, i64 p) {
            for (int i = u; i <= n; i += i & -i) {
                for (int j = v; j <= m; j += j & -j) {
                    b1[i][j] += p;
                    b2[i][j] += p * (u - 1);
                    b3[i][j] += p * (v - 1);
                    b4[i][j] += p * (u - 1) * (v - 1);
                }
            }
        };
        x++, y++;
        add(u, v, p), add(x, v, -p);
        add(x, y, p), add(u, y, -p);
    }
    i64 pointSum(int x, int y) { return matrixSum(x, y, x, y); } // 单点 [x, y] 查询
    i64 matrixSum(int u, int v, int x, int y) { // 矩形和 [u, v] ~ [x, y] 查询
        auto ask = [&](int u, int v) {
            i64 ans = 0;
            for (int i = u; i; i -= i & -i) {
                for (int j = v; j; j -= j & -j) {
                    ans += b1[i][j] * u * v;
                    ans -= b2[i][j] * v;
                    ans -= b3[i][j] * u;
                    ans += b4[i][j];
                }
            }
            return ans;
        };
        return ask(x, y) - ask(u - 1, y) - ask(x, v - 1) + ask(u - 1, v - 1);
    }
};
```

### 轻重链剖分/树链剖分

将线段树处理的部分分离，方便修改。支持链上查询/修改、子树查询/修改，建树时间复杂度 $\mathcal O(N\log N)$ ，单次查询时间复杂度 $\mathcal O(\log ^2 N)$ 。

```c++
struct Segt {
    struct node {
        int l, r, w, lazy;
    };
    vector<int> w;
    vector<node> t;
    
    Segt() {}
    #define GL (k << 1)
    #define GR (k << 1 | 1)
    
    void init(vector<int> in) {
        int n = in.size() - 1;
        w.resize(n + 1);
        for (int i = 1; i <= n; i++) {
            w[i] = in[i];
        }
        t.resize(n * 4 + 1);
        auto build = [&](auto self, int l, int r, int k = 1) {
            if (l == r) {
                t[k] = {l, r, w[l], 0}; // 如果有赋值为 0 的操作，则懒标记必须要 -1
                return;
            }
            t[k] = {l, r};
            int mid = (l + r) / 2;
            self(self, l, mid, GL);
            self(self, mid + 1, r, GR);
            pushup(k);
        };
        build(build, 1, n);
    }
    void pushdown(node &p, int lazy) { /* 【在此更新下递函数】 */
        p.w += (p.r - p.l + 1) * lazy;
        p.lazy += lazy;
    }
    void pushdown(int k) { // 不需要动
        if (t[k].lazy == 0) return;
        pushdown(t[GL], t[k].lazy);
        pushdown(t[GR], t[k].lazy);
        t[k].lazy = 0;
    }
    void pushup(int k) { // 不需要动
        auto pushup = [&](node &p, node &l, node &r) { /* 【在此更新上传函数】 */
            p.w = l.w + r.w;
        };
        pushup(t[k], t[GL], t[GR]);
    }
    void modify(int l, int r, int val, int k = 1) {
        if (l <= t[k].l && t[k].r <= r) {
            pushdown(t[k], val);
            return;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        if (l <= mid) modify(l, r, val, GL);
        if (mid < r) modify(l, r, val, GR);
        pushup(k);
    }
    int ask(int l, int r, int k = 1) {
        if (l <= t[k].l && t[k].r <= r) {
            return t[k].w;
        }
        pushdown(k);
        int mid = (t[k].l + t[k].r) / 2;
        int ans = 0;
        if (l <= mid) ans += ask(l, r, GL);
        if (mid < r) ans += ask(l, r, GR);
        return ans;
    }
};

struct HLD {
    int n, idx;
    vector<vector<int>> ver;
    vector<int> siz, dep;
    vector<int> top, son, parent;
    vector<int> in, id, val;
    Segt segt;

    HLD(int n) {
        this->n = n;
        ver.resize(n + 1);
        siz.resize(n + 1);
        dep.resize(n + 1);

        top.resize(n + 1);
        son.resize(n + 1);
        parent.resize(n + 1);

        idx = 0;
        in.resize(n + 1);
        id.resize(n + 1);
        val.resize(n + 1);
    }
    void add(int x, int y) { // 建立双向边
        ver[x].push_back(y);
        ver[y].push_back(x);
    }
    void dfs1(int x) {
        siz[x] = 1;
        dep[x] = dep[parent[x]] + 1;
        for (auto y : ver[x]) {
            if (y == parent[x]) continue;
            parent[y] = x;
            dfs1(y);
            siz[x] += siz[y];
            if (siz[y] > siz[son[x]]) {
                son[x] = y;
            }
        }
    }
    void dfs2(int x, int up) {
        id[x] = ++idx;
        val[idx] = in[x]; // 建立编号
        top[x] = up;
        if (son[x]) dfs2(son[x], up);
        for (auto y : ver[x]) {
            if (y == parent[x] || y == son[x]) continue;
            dfs2(y, y);
        }
    }
    void modify(int l, int r, int val) { // 链上修改
        while (top[l] != top[r]) {
            if (dep[top[l]] < dep[top[r]]) {
                swap(l, r);
            }
            segt.modify(id[top[l]], id[l], val);
            l = parent[top[l]];
        }
        if (dep[l] > dep[r]) {
            swap(l, r);
        }
        segt.modify(id[l], id[r], val);
    }
    void modify(int root, int val) { // 子树修改
        segt.modify(id[root], id[root] + siz[root] - 1, val);
    }
    int ask(int l, int r) { // 链上查询
        int ans = 0;
        while (top[l] != top[r]) {
            if (dep[top[l]] < dep[top[r]]) {
                swap(l, r);
            }
            ans += segt.ask(id[top[l]], id[l]);
            l = parent[top[l]];
        }
        if (dep[l] > dep[r]) {
            swap(l, r);
        }
        return ans + segt.ask(id[l], id[r]);
    }
    int ask(int root) { // 子树查询
        return segt.ask(id[root], id[root] + siz[root] - 1);
    }
    void work(auto in, int root = 1) { // 在此初始化
        assert(in.size() == n + 1);
        this->in = in;
        dfs1(root);
        dfs2(root, root);
        segt.init(val); // 建立线段树
    }
    void work(int root = 1) { // 在此初始化
        dfs1(root);
        dfs2(root, root);
        segt.init(val); // 建立线段树
    }
};
```

### 小波矩阵：静态区间第 k 小

设不同值个数为 $\sigma$，初始化复杂度 $\mathcal O(n \log n)$，空间 $\mathcal O(n \log \sigma)$，单次操作复杂度 $\mathcal O(\log \sigma)$。有==压位版本==可优化空间到 $\mathcal O(n + \tfrac{n \log n}{64})$。注意事项：
- 查找第 $k$ 小时，需要自行保证第 $k$ 小存在（$1 \le k \le R - L + 1$）；

```c++
struct WaveletMatrix {
    int lg;
    vector<i64> val;
    vector<int> mid;
    vector<vector<int>> z; // 每层的 0 前缀和

    WaveletMatrix(const vector<i64> &a) {
        int n = (int)a.size() - 1;
        val.assign(a.begin() + 1, a.end());
        sort(val.begin(), val.end());
        val.erase(unique(val.begin(), val.end()), val.end());

        int m = val.size();
        lg = m == 1 ? 1 : __lg(m - 1) + 1;

        vector<int> cur(n + 1), nxt(n + 1);
        for (int i = 1; i <= n; i++) {
            cur[i] = lower_bound(val.begin(), val.end(), a[i]) - val.begin();
        }

        mid.resize(lg);
        z.assign(lg, vector<int>(n + 1));
        for (int d = 0; d < lg; d++) {
            int pos = lg - 1 - d;
            for (int i = 1; i <= n; i++) {
                z[d][i] = z[d][i - 1] + ((cur[i] >> pos & 1) == 0);
            }

            mid[d] = z[d][n];
            int p0 = 0, p1 = mid[d];
            for (int i = 1; i <= n; i++) {
                if (cur[i] >> pos & 1) nxt[++p1] = cur[i];
                else nxt[++p0] = cur[i];
            }
            cur.swap(nxt);
        }
    }

    i64 kth(int L, int R, int k) const {
        int ans = 0;
        for (int d = 0; d < lg; d++) {
            int zL = z[d][L - 1], zR = z[d][R];
            if (k <= zR - zL) {
                L = zL + 1, R = zR;
            } else {
                k -= zR - zL, ans |= 1 << (lg - 1 - d);
                L = mid[d] + L - zL, R = mid[d] + R - zR;
            }
        }
        return val[ans];
    }

    i64 kthLarge(int L, int R, int k) const {
        return kth(L, R, R - L + 2 - k);
    }
};
```

```c++
cout << wm.kthLarge(1, 4, 2) << '\n'; // 5
```

### 普通莫队

以 $\mathcal O(N \sqrt N)$ 的复杂度完成 $Q$ 次询问的离线查询，其中每个分块的大小取 $\sqrt N=\sqrt {10^5} = 317$ ，也可以使用 `n / min<int>(n, sqrt(q))` 、 `ceil((double)n / (int)sqrt(n))` 或者 `sqrt(n)` 划分。

```c++
signed main() {
    int n;
    cin >> n;
    vector<int> w(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
    }
    
    int q;
    cin >> q;
    vector<array<int, 3>> query(q + 1);
    for (int i = 1; i <= q; i++) {
        int l, r;
        cin >> l >> r;
        query[i] = {l, r, i};
    }

        if (K[x[0]] != K[y[0]]) return x[0] < y[0];
        if (K[x[0]] & 1) return x[1] < y[1];
        return x[1] > y[1];
    });

        while (r < qr) add(w[++r]);
        while (l < ql) del(w[l++]);
        while (r > qr) del(w[r--]);
        ans[id] = val;
    }
    for (int i = 1; i <= q; i++) {
        cout << ans[i] << endl;
    }
}
```

### 分数运算类

定义了分数的四则运算，如果需要处理浮点数，那么需要将函数中的 `gcd` 运算替换为 `fgcd` 。

```c++
template<class T> struct Frac {
    T x, y;
    Frac() : Frac(0, 1) {}
    Frac(T x_) : Frac(x_, 1) {}
    Frac(T x_, T y_) : x(x_), y(y_) {
        if (y < 0) {
            y = -y;
            x = -x;
        }
    }
    
    constexpr double val() const {
        return 1. * x / y;
    }
    constexpr Frac norm() const { // 调整符号、转化为最简形式
        T p = gcd(x, y);
        return {x / p, y / p};
    }
    friend constexpr auto &operator<<(ostream &o, const Frac &j) {
        T p = gcd(j.x, j.y);
        if (j.y == p) {
            return o << j.x / p;
        } else {
            return o << j.x / p << "/" << j.y / p;
        }
    }
    constexpr Frac &operator/=(const Frac &i) {
        x *= i.y;
        y *= i.x;
        if (y < 0) {
            x = -x;
            y = -y;
        }
        return *this;
    }
    constexpr Frac &operator+=(const Frac &i) { return x = x * i.y + y * i.x, y *= i.y, *this; }
    constexpr Frac &operator-=(const Frac &i) { return x = x * i.y - y * i.x, y *= i.y, *this; }
    constexpr Frac &operator*=(const Frac &i) { return x *= i.x, y *= i.y, *this; }
    friend constexpr Frac operator+(const Frac i, const Frac j) { return i += j; }
    friend constexpr Frac operator-(const Frac i, const Frac j) { return i -= j; }
    friend constexpr Frac operator*(const Frac i, const Frac j) { return i *= j; }
    friend constexpr Frac operator/(const Frac i, const Frac j) { return i /= j; }
    friend constexpr Frac operator-(const Frac i) { return Frac(-i.x, i.y); }
    friend constexpr bool operator<(const Frac i, const Frac j) { return i.x * j.y < i.y * j.x; }
    friend constexpr bool operator>(const Frac i, const Frac j) { return i.x * j.y > i.y * j.x; }
    friend constexpr bool operator==(const Frac i, const Frac j) { return i.x * j.y == i.y * j.x; }
    friend constexpr bool operator!=(const Frac i, const Frac j) { return i.x * j.y != i.y * j.x; }
};
```

### 主席树（可持久化线段树）

```c++
struct PresidentTree {

    int cntNodes, root[N];
    struct node {
        int l, r;
        int cnt;
    } tr[4 * N + 17 * N];

        tr[u] = tr[v];
        tr[u].cnt++;
        if (l == r) return;
        int mid = (l + r) / 2;
        if (x <= mid)
            modify(tr[u].l, tr[v].l, l, mid, x);
        else
            modify(tr[u].r, tr[v].r, mid + 1, r, x);

        int mid = (l + r) / 2;
        if (k <= res)
            return kth(tr[u].l, tr[v].l, l, mid, k);

            return kth(tr[u].r, tr[v].r, mid + 1, r, k - res);
```

### KD Tree

在第 $k$ 维上的单次查询复杂度最坏为 $\mathcal O(n^{1-k^{-1}})$。

```c++
struct KDT {
    constexpr static int N = 1e5 + 10, K = 2;
    double alpha = 0.725;
    struct node {
        int info[K];
        int mn[K], mx[K];
    } tr[N];
    int ls[N], rs[N], siz[N], id[N], d[N];
    int idx, rt, cur;
    int ans;
    KDT() {
        rt = 0;
        cur = 0;
        memset(ls, 0, sizeof ls);
        memset(rs, 0, sizeof rs);
        memset(d, 0, sizeof d);
    }
    void apply(int p, int son) {
        if (son) {
            for (int i = 0; i < K; i++) {
                tr[p].mn[i] = min(tr[p].mn[i], tr[son].mn[i]);
                tr[p].mx[i] = max(tr[p].mx[i], tr[son].mx[i]);
            }
            siz[p] += siz[son];
        }
    }
    void maintain(int p) {
        for (int i = 0; i < K; i++) {
            tr[p].mn[i] = tr[p].info[i];
            tr[p].mx[i] = tr[p].info[i];
        }
        siz[p] = 1;
        apply(p, ls[p]);
        apply(p, rs[p]);
    }
    int build(int l, int r) {
        if (l > r) return 0;
        vector<double> avg(K);
        for (int i = 0; i < K; i++) {
            for (int j = l; j <= r; j++) {
                avg[i] += tr[id[j]].info[i];
            }
            avg[i] /= (r - l + 1);
        }
        vector<double> var(K);
        for (int i = 0; i < K; i++) {
            for (int j = l; j <= r; j++) {
                var[i] += (tr[id[j]].info[i] - avg[i]) * (tr[id[j]].info[i] - avg[i]);
            }
        }
        int mid = (l + r) / 2;
        int x = max_element(var.begin(), var.end()) - var.begin();
        nth_element(id + l, id + mid, id + r + 1, [&](int a, int b) {
            return tr[a].info[x] < tr[b].info[x];
        });
        d[id[mid]] = x;
        ls[id[mid]] = build(l, mid - 1);
        rs[id[mid]] = build(mid + 1, r);
        maintain(id[mid]);
        return id[mid];
    }
    void print(int p) {
        if (!p) return;
        print(ls[p]);
        id[++idx] = p;
        print(rs[p]);
    }
    void rebuild(int &p) {
        idx = 0;
        print(p);
        p = build(1, idx);
    }
    bool bad(int p) {
        return alpha * siz[p] <= max(siz[ls[p]], siz[rs[p]]);
    }
    void insert(int &p, int cur) {
        if (!p) {
            p = cur;
            maintain(p);
            return;
        }
        if (tr[p].info[d[p]] > tr[cur].info[d[p]]) insert(ls[p], cur);
        else insert(rs[p], cur);
        maintain(p);
        if (bad(p)) rebuild(p);
    }
    void insert(vector<int> &a) {
        cur++;
        for (int i = 0; i < K; i++) {
            tr[cur].info[i] = a[i];
        }
        insert(rt, cur);
    }
    bool out(int p, vector<int> &a) {
        for (int i = 0; i < K; i++) {
            if (a[i] < tr[p].mn[i]) {
                return true;
            }
        }
        return false;
    }
    bool in(int p, vector<int> &a) {
        for (int i = 0; i < K; i++) {
            if (a[i] < tr[p].info[i]) {
                return false;
            }
        }
        return true;
    }
    bool all(int p, vector<int> &a) {
        for (int i = 0; i < K; i++) {
            if (a[i] < tr[p].mx[i]) {
                return false;
            }
        }
        return true;
    }
    void query(int p, vector<int> &a) {
        if (!p) return;
        if (out(p, a)) return;
        if (all(p, a)) {
            ans += siz[p];
            return;
        }
        if (in(p, a)) ans++;
        query(ls[p], a);
        query(rs[p], a);
    }
    int query(vector<int> &a) {
        ans = 0;
        query(rt, a);
        return ans;
    }
};
```

### 线性基（高斯消元法）

设向量长度为 $N$（一般取 $63$），总数为 $M$，时间复杂度为 $\mathcal O(NM)$ 。

```c++
struct LB { // Linear Basis
    using i64 = long long;
    const int BASE = 63;
    vector<i64> d, p;
    int cnt, flag;

    LB() {
        d.resize(BASE + 1);
        p.resize(BASE + 1);
        cnt = flag = 0;
    }
    bool insert(i64 val) {
        for (int i = BASE - 1; i >= 0; i--) {
            if (val & (1ll << i)) {
                if (!d[i]) {
                    d[i] = val;
                    return true;
                }
                val ^= d[i];
            }
        }
        flag = 1; //可以异或出0
        return false;
    }
    bool check(i64 val) { // 判断 val 是否能被异或得到
        for (int i = BASE - 1; i >= 0; i--) {
            if (val & (1ll << i)) {
                if (!d[i]) {
                    return false;
                }
                val ^= d[i];
            }
        }
        return true;
    }
    i64 ask_max() {
        i64 res = 0;
        for (int i = BASE - 1; i >= 0; i--) {
            if ((res ^ d[i]) > res) res ^= d[i];
        }
        return res;
    }
    i64 ask_min() {
        if (flag) return 0; // 特判 0
        for (int i = 0; i <= BASE - 1; i++) {
            if (d[i]) return d[i];
        }
    }
    void rebuild() { // 第k小值独立预处理
        for (int i = BASE - 1; i >= 0; i--) {
            for (int j = i - 1; j >= 0; j--) {
                if (d[i] & (1ll << j)) d[i] ^= d[j];
            }
        }
        for (int i = 0; i <= BASE - 1; i++) {
            if (d[i]) p[cnt++] = d[i];
        }
    }
    i64 kthquery(i64 k) { // 查询能被异或得到的第 k 小值, 如不存在则返回 -1
        if (flag) k--; // 特判 0, 如果不需要 0, 直接删去
        if (!k) return 0;
        i64 res = 0;
        if (k >= (1ll << cnt)) return -1;
        for (int i = BASE - 1; i >= 0; i--) {
            if (k & (1LL << i)) res ^= p[i];
        }
        return res;
    }
    void Merge(const LB &b) { // 合并两个线性基
        for (int i = BASE - 1; i >= 0; i--) {
            if (b.d[i]) {
                insert(b.d[i]);
            }
        }
    }
};
```

### 珂朵莉树 (OD Tree)

区间赋值的数据结构都可以骗分，在数据随机的情况下，复杂度可以保证，时间复杂度：$\mathcal O(N\log\log N)$ 。

```c++
struct ODT {
    struct node {
        int l, r;
        mutable LL v;
        node(int l, int r = -1, LL v = 0) : l(l), r(r), v(v) {}
        bool operator<(const node &o) const {
            return l < o.l;
        }
    };
    set<node> s;
    ODT() {
        s.clear();
    }
    auto split(int pos) {
        auto it = s.lower_bound(node(pos));
        if (it != s.end() && it->l == pos) return it;
        it--;
        int l = it->l, r = it->r;
        LL v = it->v;
        s.erase(it);
        s.insert(node(l, pos - 1, v));
        return s.insert(node(pos, r, v)).first;
    }
    void assign(int l, int r, LL x) {
        auto itr = split(r + 1), itl = split(l);
        s.erase(itl, itr);
        s.insert(node(l, r, x));
    }
    void add(int l, int r, LL x) {
        auto itr = split(r + 1), itl = split(l);
        for (auto it = itl; it != itr; it++) {
            it->v += x;
        }
    }
    LL kth(int l, int r, int k) {
        vector<pair<LL, int>> a;
        auto itr = split(r + 1), itl = split(l);
        for (auto it = itl; it != itr; it++) {
            a.push_back(pair<LL, int>(it->v, it->r - it->l + 1));
        }
        sort(a.begin(), a.end());
        for (auto [val, len] : a) {
            k -= len;
            if (k <= 0) return val;
        }
    }
    LL power(LL a, int b, int mod) {
        a %= mod;
        LL res = 1;
        for (; b; b /= 2, a = a * a % mod) {
            if (b % 2) {
                res = res * a % mod;
            }
        }
        return res;
    }
    LL powersum(int l, int r, int x, int mod) {
        auto itr = split(r + 1), itl = split(l);
        LL ans = 0;
        for (auto it = itl; it != itr; it++) {
            ans = (ans + power(it->v, x, mod) * (it->r - it->l + 1) % mod) % mod;
        }
        return ans;
    }
};
```

### 大整数类（高精度计算）

```c++

const int base_digits = 9; // 分解为九个数位一个数字
struct bigint {

    bigint() : sign(1) {}
    bigint operator-() const {
        bigint res = *this;
        res.sign = -sign;
        return res;
    }

    }
    void operator=(const bigint &v) {

        a = v.a;
    }
    void operator=(long long v) {
        a.clear();
        sign = 1;
        if (v < 0) sign = -1, v = -v;
        for (; v > 0; v = v / base) {
            a.push_back(v % base);
        }
    }

    // 基础加减乘除
    bigint operator+(const bigint &v) const {
        if (sign == v.sign) {
            bigint res = v;
            for (int i = 0, carry = 0; i < (int)max(a.size(), v.a.size()) || carry; ++i) {
                if (i == (int)res.a.size()) {
                    res.a.push_back(0);
                }
                res.a[i] += carry + (i < (int)a.size() ? a[i] : 0);
                carry = res.a[i] >= base;
                if (carry) {
                    res.a[i] -= base;
                }
            }
            return res;
        }
        return *this - (-v);
    }
    bigint operator-(const bigint &v) const {
        if (sign == v.sign) {
            if (abs() >= v.abs()) {
                bigint res = *this;
                for (int i = 0, carry = 0; i < (int)v.a.size() || carry; ++i) {
                    res.a[i] -= carry + (i < (int)v.a.size() ? v.a[i] : 0);
                    carry = res.a[i] < 0;
                    if (carry) {
                        res.a[i] += base;
                    }
                }
                res.trim();
                return res;
            }
            return -(v - *this);
        }
        return *this + (-v);
    }
    void operator*=(int v) {
        check(v);
        for (int i = 0, carry = 0; i < (int)a.size() || carry; ++i) {
            if (i == (int)a.size()) {
                a.push_back(0);
            }
            long long cur = a[i] * (long long)v + carry;
            carry = (int)(cur / base);
            a[i] = (int)(cur % base);
        }
        trim();
    }
    void operator/=(int v) {
        check(v);
        for (int i = (int)a.size() - 1, rem = 0; i >= 0; --i) {
            long long cur = a[i] + rem * (long long)base;
            a[i] = (int)(cur / v);
            rem = (int)(cur % v);
        }
        trim();
    }

        if (v < 0) {
            v = -v;
        }

        for (int i = a.size() - 1; i >= 0; --i) {
            m = (a[i] + m * (long long)base) % v;
        }

        for (int i = a.size() - 1; i >= 0; i--)

        vll a(a6.begin(), a6.end());
        vll b(b6.begin(), b6.end());
        while (a.size() < b.size()) a.push_back(0);
        while (b.size() < a.size()) b.push_back(0);
        while (a.size() & (a.size() - 1)) a.push_back(0), b.push_back(0);
        vll c = karatsubaMultiply(a, b);
        bigint res;
        res.sign = sign * v.sign;
        for (int i = 0, carry = 0; i < (int)c.size(); i++) {
            long long cur = c[i] + carry;
            res.a.push_back((int)(cur % 1000000));
            carry = (int)(cur / 1000000);
        }
        res.a = convert_base(res.a, 6, base_digits);
        res.trim();
        return res;
    }
    friend pair<bigint, bigint> divmod(const bigint &a1,
                                       const bigint &b1) { // 大整数除大整数，同时返回答案与余数
        int norm = base / (b1.a.back() + 1);
        bigint a = a1.abs() * norm;
        bigint b = b1.abs() * norm;
        bigint q, r;
        q.a.resize(a.a.size());
        for (int i = a.a.size() - 1; i >= 0; i--) {
            r *= base;
            r += a.a[i];
            int s1 = r.a.size() <= b.a.size() ? 0 : r.a[b.a.size()];
            int s2 = r.a.size() <= b.a.size() - 1 ? 0 : r.a[b.a.size() - 1];
            int d = ((long long)base * s1 + s2) / b.a.back();
            r -= b * d;
            while (r < 0) r += b, --d;
            q.a[i] = d;
        }
        q.sign = a1.sign * b1.sign;
        r.sign = a1.sign;
        q.trim();
        r.trim();
        return make_pair(q, r / norm);
    }
    static vector<int> convert_base(const vector<int> &a, int old_digits, int new_digits) {
        vector<long long> p(max(old_digits, new_digits) + 1);
        p[0] = 1;
        for (int i = 1; i < (int)p.size(); i++) p[i] = p[i - 1] * 10;
        vector<int> res;
        long long cur = 0;
        int cur_digits = 0;
        for (int i = 0; i < (int)a.size(); i++) {
            cur += a[i] * p[cur_digits];
            cur_digits += old_digits;
            while (cur_digits >= new_digits) {
                res.push_back((int)(cur % p[new_digits]));
                cur /= p[new_digits];
                cur_digits -= new_digits;
            }
        }
        res.push_back((int)cur);
        while (!res.empty() && !res.back()) res.pop_back();
        return res;
    }
    static vll karatsubaMultiply(const vll &a, const vll &b) {
        int n = a.size();
        vll res(n + n);
        if (n <= 32) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    res[i + j] += a[i] * b[j];
                }
            }
            return res;
        }

        int k = n >> 1;
        vll a1(a.begin(), a.begin() + k);
        vll a2(a.begin() + k, a.end());
        vll b1(b.begin(), b.begin() + k);
        vll b2(b.begin() + k, b.end());

        vll a1b1 = karatsubaMultiply(a1, b1);
        vll a2b2 = karatsubaMultiply(a2, b2);

        for (int i = 0; i < k; i++) a2[i] += a1[i];
        for (int i = 0; i < k; i++) b2[i] += b1[i];

        vll r = karatsubaMultiply(a2, b2);
        for (int i = 0; i < (int)a1b1.size(); i++) r[i] -= a1b1[i];
        for (int i = 0; i < (int)a2b2.size(); i++) r[i] -= a2b2[i];

        for (int i = 0; i < (int)r.size(); i++) res[i + k] += r[i];
        for (int i = 0; i < (int)a1b1.size(); i++) res[i] += a1b1[i];
        for (int i = 0; i < (int)a2b2.size(); i++) res[i + n] += a2b2[i];
        return res;
    }

    void operator*=(const bigint &v) {
        *this = *this * v;
    }
    bigint operator/(const bigint &v) const {
        return divmod(*this, v).first;
    }
    void operator/=(const bigint &v) {
        *this = *this / v;
    }
    bigint operator%(const bigint &v) const {
        return divmod(*this, v).second;
    }
    void operator%=(const bigint &v) {
        *this = *this % v;
    }
};
```

### 常见结论

题意：（区间移位问题）要求将整个序列左移/右移若干个位置，例如，原序列为 $A=(a_1, a_2, \dots, a_n)$ ，右移 $x$ 位后变为 $A=(a_{x+1}, a_{x+2}, \dots, a_n,a_1,a_2,\dots, a_x)$ 。

区间的端点只是一个数字，即使被改变了，通过一定的转换也能够还原，所以我们可以 $\mathcal O(1)$ 解决这一问题。为了方便计算，我们规定下标从 $0$ 开始，即整个线段的区间为 $[0, n)$ ，随后，使用一个偏移量 `shift` 记录。使用 `shift = (shift + x) % n;` 更新偏移量；此后的区间查询/修改前，再将坐标偏移回去即可，下方代码使用区间修改作为示例。

```c++
cin >> l >> r >> x;
l--; // 坐标修改为 0 开始
r--;
l = (l + shift) % n; // 偏移
r = (r + shift) % n;
if (l > r) { // 区间分离则分别操作
    segt.modify(l, n - 1, x);
    segt.modify(0, r, x);
} else {
    segt.modify(l, r, x);
}
```

### 常见例题

题意：（带修莫队 - 维护队列）要求能够处理以下操作：

- `'Q' l r` ：询问区间 $[l,r]$ 有几个颜色；
- `'R' idx w` ：将下标 $\tt idx$ 的颜色修改为 $\tt w$ 。

输入格式为：第一行 $n$ 和 $q\ (1\le n, q\le 133333)$ 分别代表区间长度和操作数量；第二行 $n$ 个整数 $a_1,a_2\dots,a_n\ (1\le a_i\le 10^6)$ 代表初始颜色；随后 $q$ 行为具体操作。

```c++
const int N = 1e6 + 7;
signed main() {
    int n, q;
    cin >> n >> q;
    vector<int> w(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
    }
    
    vector<array<int, 4>> query = {{}}; // {左区间, 右区间, 累计修改次数, 下标}
    vector<array<int, 2>> modify = {{}}; // {修改的值, 修改的元素下标}
    for (int i = 1; i <= q; i++) {
        char op;
        cin >> op;
        if (op == 'Q') {
            int l, r;
            cin >> l >> r;
            query.push_back({l, r, (int)modify.size() - 1, (int)query.size()});
        } else {
            int idx, w;
            cin >> idx >> w;
            modify.push_back({w, idx});
        }
    }
    
    int Knum = 2154; // 计算块长
    vector<int> K(n + 1);
    for (int i = 1; i <= n; i++) { // 固定块长
        K[i] = (i - 1) / Knum + 1;
    }
    sort(query.begin() + 1, query.end(), [&](auto x, auto y) {
        if (K[x[0]] != K[y[0]]) return x[0] < y[0];
        if (K[x[1]] != K[y[1]]) return x[1] < y[1];
        return x[3] < y[3];
    });
    
    int l = 1, r = 0, val = 0;
    int t = 0; // 累计修改次数
    vector<int> ans(query.size()), cnt(N);
    for (int i = 1; i < query.size(); i++) {
        auto [ql, qr, qt, id] = query[i];
        auto add = [&](int x) -> void {
            if (cnt[x] == 0) ++ val;
            ++ cnt[x];
        };
        auto del = [&](int x) -> void {
            -- cnt[x];
            if (cnt[x] == 0) -- val;
        };
        auto time = [&](int x, int l, int r) -> void {
            if (l <= modify[x][1] && modify[x][1] <= r) { //当修改的位置在询问期间内部时才会改变num的值
                del(w[modify[x][1]]);
                add(modify[x][0]);
            }
            swap(w[modify[x][1]], modify[x][0]); //直接交换修改数组的值与原始值，减少额外的数组开销，且方便复原
        };
        while (l > ql) add(w[--l]);
        while (r < qr) add(w[++r]);
        while (l < ql) del(w[l++]);
        while (r > qr) del(w[r--]);
        while (t < qt) time(++t, ql, qr);
        while (t > qt) time(t--, ql, qr);
        ans[id] = val;
    }
    for (int i = 1; i < ans.size(); i++) {
        cout << ans[i] << endl;
    }
}
```
