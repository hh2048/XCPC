## 第八章：数据结构

约定：
- 数组下标、元素排名等均从 $1$ 开始（第 $0$ 项/行/列默认作为补零哨兵）；
- 查询区间为闭区间 $[l, r]$；
- 相同名称的封装需独立选择，不能同时复制。

模板非百科全书，如存在多个数据结构可以维护同一信息，默认仅保留复杂度最优的版本。

### 前缀和与差分：离线区间修改、静态区间查询

空间与初始化复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(1)$。常见操作为区间和、区间异或和。

```c++
vector<i64> d(n + 2);
i64 last = 0;
for (int i = 1; i <= n; i++) { // 由初始数组 a 构造差分
    d[i] = a[i] - last;
    last = a[i];
}
auto rangeAdd = [&](int l, int r, i64 v) {
    d[l] += v;
    d[r + 1] -= v;
};

// 所有修改完成后还原 a
for (int i = 1; i <= n; i++) {
    d[i] += d[i - 1];
    a[i] = d[i];
}

vector<i64> pre(n + 1);
for (int i = 1; i <= n; i++) {
    pre[i] = pre[i - 1] + a[i];
}
auto rangeSum = [&](int l, int r) {
    return pre[r] - pre[l - 1];
};
```

### 二维前缀和与差分：离线矩形修改、静态矩形查询

空间与初始化复杂度 $\mathcal O(nm)$，单次操作复杂度 $\mathcal O(1)$。

```c++
vector<vector<i64>> d(n + 2, vector<i64>(m + 2));
for (int x = 1; x <= n; x++) {
    for (int y = 1; y <= m; y++) { // 由初始矩阵 a 构造二维差分
        d[x][y] = a[x][y] - a[x - 1][y] - a[x][y - 1] + a[x - 1][y - 1];
    }
}
auto matrixAdd = [&](int u, int v, int x, int y, i64 p) { // [u, v] ~ [x, y] 全部加 p
    x++, y++;
    d[u][v] += p, d[x][v] -= p;
    d[x][y] += p, d[u][y] -= p;
};

// 所有修改完成后还原 a
for (int x = 1; x <= n; x++) {
    for (int y = 1; y <= m; y++) {
        d[x][y] += d[x - 1][y] + d[x][y - 1] - d[x - 1][y - 1];
        a[x][y] = d[x][y];
    }
}

vector<vector<i64>> pre(n + 1, vector<i64>(m + 1));
for (int x = 1; x <= n; x++) {
    for (int y = 1; y <= m; y++) {
        pre[x][y] = a[x][y] + pre[x - 1][y] + pre[x][y - 1] - pre[x - 1][y - 1];
    }
}
auto matrixSum = [&](int u, int v, int x, int y) {
    return pre[x][y] - pre[u - 1][y] - pre[x][v - 1] + pre[u - 1][v - 1];
};
```

### ST 表：静态区间查询

用于静态数组的幂等区间查询（$x \text{ 运算符 } x=x$），**不支持修改操作**。空间与初始化复杂度 $\mathcal O(n \log n)$，单次操作复杂度 $\mathcal O(1)$。

`Op` 为无状态、可默认构造的函数对象，按左区间、右区间的顺序合并，必须满足结合律与幂等律。常见操作为区间最小值（$\min(x,x)=x$）、最大值（$\max(x,x)=x$）、$\gcd$（$\gcd(x,x)=x$）、按位与、按位或。

```c++
template<class Op> struct ST {
    int n;
    vector<vector<i64>> st;

    ST(const vector<i64> &a) : n(int(a.size()) - 1) {
        st.push_back(a);
        for (int k = 1; (1 << k) <= n; k++) {
            int len = 1 << k, half = len >> 1;
            st.emplace_back(n - len + 2);
            for (int l = 1; l + len - 1 <= n; l++) {
                st[k][l] = Op{}(st[k - 1][l], st[k - 1][l + half]);
            }
        }
    }
    i64 query(int l, int r) const { // 要求 1 <= l <= r <= n
        int k = __lg(r - l + 1);
        return Op{}(st[k][l], st[k][r - (1 << k) + 1]);
    }
};
```

### 树状数组：单点修改、区间查询

空间复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$。

#### 单点加、区间和

```c++
struct BIT {
    int n;
    vector<i64> w;
    BIT(int n) : n(n), w(n + 1) {}

    void pointAdd(int x, i64 v) { // 单点 x 修改
        for (; x <= n; x += x & -x) {
            w[x] += v;
        }
    }
    i64 pointSum(int x) { return rangeSum(x, x); } // 单点 x 查询
    i64 rangeSum(int l, int r) { // 区间和 [l, r] 查询
        auto sum = [&](int x) {
            i64 ans = 0;
            for (; x; x -= x & -x) {
                ans += w[x];
            }
            return ans;
        };
        return sum(r) - sum(l - 1);
    }
};
```

#### 双树状数组：区间加、区间和

常数比上一个版本略微大一些。

```c++
struct RangeBIT {
    int n;
    vector<i64> a, b;
    RangeBIT(int n) : n(n), a(n + 1), b(n + 1) {}
    
    void pointAdd(int x, i64 v) { rangeAdd(x, x, v); } // 单点 x 修改
    void rangeAdd(int l, int r, i64 v) { // 区间 [l, r] 修改
        auto add = [&](int x, i64 v) {
            for (int i = x; i <= n; i += i & -i) {
                a[i] += v;
                b[i] += x * v;
            }
        };
        add(l, v), add(r + 1, -v);
    }
    i64 pointSum(int x) { return rangeSum(x, x); } // 单点 x 查询
    i64 rangeSum(int l, int r) { // 区间和 [l, r] 查询
        auto sum = [&](int x) {
            i64 ans = 0;
            for (int i = x; i; i -= i & -i) {
                ans += (x + 1) * a[i] - b[i];
            }
            return ans;
        };
        return sum(r) - sum(l - 1);
    }
};
```

### 并查集

除可撤销并查集外，均支持路径压缩和按大小合并，均摊复杂度为 $\mathcal O(\alpha(n))$（比 $\log n$ 更小，在 $n = 2 \times 10^5$ 时，$\alpha(n) < 5$）。根编号不固定。

#### 常用封装

```c++
struct DSU {
    vector<int> fa, siz;
    int cnt; // 当前连通块数量

    DSU(int n) : fa(n + 1), siz(n + 1, 1), cnt(n) {
        iota(fa.begin(), fa.end(), 0);
    }
    int get(int x) {
        while (x != fa[x]) {
            x = fa[x] = fa[fa[x]];
        }
        return x;
    }
    bool merge(int x, int y) { // 返回是否合并两个连通块
        x = get(x), y = get(y);
        if (x == y) return false;
        if (siz[x] < siz[y]) swap(x, y);
        fa[y] = x;
        siz[x] += siz[y];
        cnt--;
        return true;
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    int size(int x) { // 返回 x 所在的连通块中点的数量
        return siz[get(x)];
    }
};
```

#### 扩展封装（维护无向图连通块信息）

仅用于无向图，可额外维护无向图连通块数量、点数和边数（重边与自环正常计数）。

```c++
struct DSU {
    vector<int> fa, siz, edges;
    int cnt; // 连通块数量

    DSU(int n) : fa(n + 1), siz(n + 1, 1), edges(n + 1), cnt(n) {
        iota(fa.begin(), fa.end(), 0);
    }
    int get(int x) {
        while (x != fa[x]) {
            x = fa[x] = fa[fa[x]];
        }
        return x;
    }
    bool add(int x, int y) { // 加入一条无向边，返回是否合并两个连通块
        x = get(x), y = get(y);
        if (x == y) {
            edges[x]++;
            return false;
        }
        if (siz[x] < siz[y]) swap(x, y);
        fa[y] = x;
        siz[x] += siz[y];
        edges[x] += edges[y] + 1;
        cnt--;
        return true;
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    int size(int x) { // 返回 x 所在的连通块中点的数量
        return siz[get(x)];
    }
    int edgeCount(int x) { // 返回 x 所在的连通块中边的数量
        return edges[get(x)];
    }
    bool isTree(int x) { // 判断 x 所在的连通块是否为树
        x = get(x);
        return edges[x] == siz[x] - 1;
    }
};
```

#### 带权并查集

当题目逐条给出两点间的差值或关系，并要求判断新陈述能否与已有陈述同时成立时使用。构造时 `mod = 0` 表示普通整数差值；`mod = 2` 可处理奇偶 / 二分图关系，`mod = 3` 可处理食物链等循环关系。

```c++
struct wDSU {
    vector<int> fa, siz;
    vector<i64> dis; // val[x] - val[fa[x]]
    i64 mod;

    wDSU(int n, i64 M = 0) : fa(n + 1), siz(n + 1, 1), dis(n + 1), mod(M) {
        iota(fa.begin(), fa.end(), 0);
    }
    i64 norm(i64 x) const {
        if (!mod) return x;
        x %= mod;
        return x < 0 ? x + mod : x;
    }
    int get(int x) {
        if (x == fa[x]) return x;
        int p = fa[x];
        fa[x] = get(p);
        dis[x] = norm(dis[x] + dis[p]);
        return fa[x];
    }
    bool add(int x, int y, i64 w) { // val[x] - val[y] = w，返回新陈述是否相容
        int rx = get(x), ry = get(y);
        w = norm(w);
        if (rx == ry) return norm(dis[x] - dis[y]) == w;
        i64 rootDiff = norm(w - dis[x] + dis[y]); // val[rx] - val[ry]
        if (siz[rx] < siz[ry]) {
            fa[rx] = ry;
            dis[rx] = rootDiff;
            siz[ry] += siz[rx];
        } else {
            fa[ry] = rx;
            dis[ry] = norm(-rootDiff);
            siz[rx] += siz[ry];
        }
        return true; // 注意，即使两点已经连通，只要关系正确仍返回 true
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    i64 diff(int x, int y) { // 要求 same(x, y)
        get(x), get(y);
        return norm(dis[x] - dis[y]);
    }
};
```

#### 可撤销并查集

不能路径压缩，仅成功合并会写入历史，`get`、`same`、`merge` 的复杂度为 $\mathcal O(\log n)$，一次 `revert(t)` 的复杂度与撤销的成功合并次数成正比。

```c++
struct rDSU {
    vector<int> fa, siz;
    vector<array<int, 2>> his;
    int cnt;

    rDSU(int n) : fa(n + 1), siz(n + 1, 1), cnt(n) {
        iota(fa.begin(), fa.end(), 0);
    }
    int get(int x) { // 不能路径压缩
        while (x != fa[x]) x = fa[x];
        return x;
    }
    bool merge(int x, int y) {
        x = get(x), y = get(y);
        if (x == y) return false;
        if (siz[x] < siz[y]) swap(x, y);
        his.push_back({x, y}); // 仅记录成功合并
        fa[y] = x;
        siz[x] += siz[y];
        cnt--;
        return true;
    }
    int time() const { // 返回当前快照数量
        return his.size();
    }
    void revert(int t) { // 恢复到时间戳 t 的快照
        while (time() > t) {
            auto [x, y] = his.back();
            his.pop_back();
            fa[y] = y;
            siz[x] -= siz[y];
            cnt++;
        }
    }
    bool same(int x, int y) {
        return get(x) == get(y);
    }
    int size(int x) {
        return siz[get(x)];
    }
};
```

<blockquote>

例如，《P3402. 【模板】可持久化并查集》构建一个离线可持久化并查集，每个操作以某个旧版本（`op[u]`）为基础，先将版本依赖建成树（`son[u]`）。

```c++

    int t = dsu.time(); // 保存一次快照
    if (op[u].type == 0)  dsu.merge(op[u].x, op[u].y);
    if (op[u].type == 1)  ans[u] = dsu.same(op[u].x, op[u].y);
    for (int v : son[u]) {
        dfs(v);
    }
    dsu.revert(t); // 恢复到所保存的快照
}
```

### 静态线段树：单点修改、静态区间查询

适用于区间信息能由左右子区间合并、修改只落在单个位置的题目。空间与初始化复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$。

使用 zkw 风格，自底向上迭代封装，常数更优。

#### 统一封装（Info）

需要定义两项内容：
- `Info` 存储一个区间的信息，且 `Info()` 表示空区间；
- `operator+` 按“左区间 + 右区间”合并，必须满足结合律，**不要求交换律**。

```c++
template<class Info> struct SegTree {
    int n = 0;
    vector<Info> t;

    SegTree(const vector<Info> &a) {
        n = int(a.size()) - 1;
        t.assign(2 * n, Info());
        for (int i = 1; i <= n; i++) {
            t[n + i - 1] = a[i];
        }
        for (int p = n - 1; p; p--) {
            pull(p);
        }
    }
    void pull(int p) { t[p] = t[p << 1] + t[p << 1 | 1]; }
    void pointSet(int x, const Info &v) { // 单点 x 赋值为 v
        t[x += n - 1] = v;
        for (x >>= 1; x; x >>= 1) {
            pull(x);
        }
    }
    Info rangeQuery(int l, int r) const { // 查询闭区间 [l, r]
        Info u, v;
        for (l += n - 1, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) u = u + t[l++]; // 若信息不满足交换律（矩阵乘法、字符串拼接），必须注意顺序
            if (r & 1) v = t[--r] + v; // 若信息不满足交换律（矩阵乘法、字符串拼接），必须注意顺序
        }
        return u + v;
    }
};
```

最常见的使用方法即区间和，但是使用树状数组更佳，这里仅展示用法：

再例如区间最小/最大值查询。

维护多个查询参数时，推荐补全构造函数，使用起来会更方便一些。

```c++
struct Info {
    i64 mn = numeric_limits<i64>::max();
    i64 mx = numeric_limits<i64>::lowest();

    Info() = default;
    Info(i64 x) : mn(x), mx(x) {} // 区间维护多参数时，这个构造函数允许使用简单的单值初始化叶子节点
    Info(i64 lo, i64 hi) : mn(lo), mx(hi) {} // 这个构造函数允许下面加法操作时用大括号初始化一个 Info 对象
};
Info operator+(const Info &a, const Info &b) {
    return {min(a.mn, b.mn), max(a.mx, b.mx)}; // 使用大括号初始化一个 Info 对象
}

vector<Info> a = {{}, {1}, {2}, {3}, {4}, {5}}; // 单值初始化叶子节点
```

#### 常见查询速查

| 查询内容 | 成员默认值，即 `Info()` | 左右合并 | 备注 |
| :---: | --- | :---: | :---: |
| 区间和 | `sum = 0` | 加法 | 直接使用树状数组更佳 |
| 区间最小值 | `mn = numeric_limits<i64>::max()` | `min` |
| 区间最大值 | `mx = numeric_limits<i64>::lowest()` | `max` |
| 区间 gcd | `g = 0` | `gcd` |
| 区间异或 | `xr = 0` | 按位异或 |
| 区间按位或 | `mask = 0` | 按位或 |
| 区间按位与 | `mask = ~i64(0)` | 按位与 |

`gcd(0, x) = |x|`，因此 $0$ 是 gcd 的单位元。

最大值的单位元必须使用 `lowest()`；`min()` 并不表示最小可表示值。

#### 区间最大子段和

维护非空最大子段和。`sum` 为区间和，`lmx`、`rmx` 分别为最大前缀和、最大后缀和，`mx` 为最大子段和。

```c++
struct Info {
    static constexpr i64 P = -(1LL << 60); // 必须小于所有可能的非空子段和
    i64 sum = 0;
    i64 lmx = P, rmx = P, mx = P;

    Info() = default;
    Info(i64 x) : sum(x), lmx(x), rmx(x), mx(x) {}
    Info(i64 s, i64 l, i64 r, i64 m) : sum(s), lmx(l), rmx(r), mx(m) {}
};
Info operator+(const Info &a, const Info &b) {
    return {
        a.sum + b.sum,
        max(a.lmx, a.sum + b.lmx),
        max(b.rmx, b.sum + a.rmx),
        max({a.mx, b.mx, a.rmx + b.lmx})
    };
}

vector<Info> a = {{}, {-2}, {3}, {-1}, {4}, {-5}};
SegTree<Info> t(a);

cout << t.rangeQuery(1, 5).mx << "\n"; // 6
t.pointSet(1, {5});
cout << t.rangeQuery(1, 5).mx << "\n"; // 11
```

### 懒标记线段树：区间修改、区间查询

适用于区间修改能在 $\mathcal O(1)$ 内直接更新整个区间信息的题目。整体复杂度与不带懒标记的线段树封装相同，空间与初始化复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$，但常数更大。

使用标准递归风格，常数略大，但是兼容性更好。

#### 统一封装（Info + Tag）

需要额外定义：
- `Tag::apply(v)`：将新操作 `v` 叠加到已有标记之后，下传时等价于先执行旧标记，再执行 `v`。
- `Info::apply(v, len)`：将标记 `v` 作用到长度为 `len` 的区间信息。

封装内部用 `has` 判断标记是否存在，`Tag` 无需用特殊值表示空操作。

```c++
template<class Info, class Tag> struct LazySegTree {
    int n;
    vector<Info> info;
    vector<Tag> tag;
    vector<char> has; // 用于判断 tag[p] 是否有效

    LazySegTree(const vector<Info> &a) {
        n = int(a.size()) - 1;
        info.assign(4 * n + 4, Info());
        tag.resize(4 * n + 4);
        has.assign(4 * n + 4, 0);
        build(1, 1, n, a);
    }
    void pointSet(int x, const Info &v) { // 单点 x 赋值为 v
        pointSet(1, 1, n, x, v);
    }
    void rangeApply(int l, int r, const Tag &v) { // 区间 [l, r] 修改
        rangeApply(1, 1, n, l, r, v);
    }
    Info rangeQuery(int l, int r) { // 查询区间 [l, r]
        return rangeQuery(1, 1, n, l, r);
    }

private:
    void build(int p, int l, int r, const vector<Info> &a) {
        if (l == r) return info[p] = a[l], void();
        int m = (l + r) / 2;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        pull(p);
    }
    void pull(int p) {
        info[p] = info[p << 1] + info[p << 1 | 1];
    }
    void apply(int p, int l, int r, const Tag &v) {
        info[p].apply(v, r - l + 1);
        if (has[p]) tag[p].apply(v);
        else tag[p] = v, has[p] = 1;
    }
    void push(int p, int l, int r) {
        if (!has[p] || l == r) return;
        int m = (l + r) / 2;
        apply(p << 1, l, m, tag[p]);
        apply(p << 1 | 1, m + 1, r, tag[p]);
        has[p] = 0;
    }
    void pointSet(int p, int l, int r, int x, const Info &v) {
        if (l == r) return info[p] = v, has[p] = 0, void();
        push(p, l, r);
        int m = (l + r) / 2;
        if (x <= m) pointSet(p << 1, l, m, x, v);
        else pointSet(p << 1 | 1, m + 1, r, x, v);
        pull(p);
    }
    void rangeApply(int p, int l, int r, int x, int y, const Tag &v) {
        if (x <= l && r <= y) return apply(p, l, r, v);
        push(p, l, r);
        int m = (l + r) / 2;
        if (x <= m) rangeApply(p << 1, l, m, x, y, v);
        if (m < y) rangeApply(p << 1 | 1, m + 1, r, x, y, v);
        pull(p);
    }
    Info rangeQuery(int p, int l, int r, int x, int y) {
        if (x <= l && r <= y) return info[p];
        push(p, l, r);
        int m = (l + r) / 2;
        if (y <= m) return rangeQuery(p << 1, l, m, x, y);
        if (m < x) return rangeQuery(p << 1 | 1, m + 1, r, x, y);
        return rangeQuery(p << 1, l, m, x, y) + rangeQuery(p << 1 | 1, m + 1, r, x, y);
    }
};
```

维护区间和查询时，需要额外维护 `B` 个计数器，统计每一位为 $1$ 的元素个数。设所有初值和异或值满足 $0 \le x < 2^B$。

空间与初始化复杂度 $\mathcal O(Bn)$，单次操作复杂度 $\mathcal O(B\log n)$。

```c++
constexpr int B = 31; // 保证所有数都在 [0, 2^B) 内

struct Tag {
    i64 mask = 0;
    void apply(const Tag &v) { mask ^= v.mask; }
};
struct Info {
    array<int, B> cnt{};
    
    Info() = default;
    Info(i64 x) {
        for (int i = 0; i < B; i++) {
            cnt[i] = x >> i & 1;
        }
    }
    void apply(const Tag &v, int len) {
        for (int i = 0; i < B; i++) {
            if (v.mask >> i & 1) {
                cnt[i] = len - cnt[i];
            }
        }
    }
    i64 sum() const {
        i64 res = 0;
        for (int i = 0; i < B; i++) {
            res += 1LL * cnt[i] * (1LL << i);
        }
        return res;
    }
};
Info operator+(const Info &a, const Info &b) {
    Info c;
    for (int i = 0; i < B; i++) {
        c.cnt[i] = a.cnt[i] + b.cnt[i];
    }
    return c;
}

vector<Info> a = {{}, {1}, {2}, {3}, {4}, {5}};
LazySegTree<Info, Tag> t(a);

t.rangeApply(2, 4, {3}); // [1, 1, 0, 7, 5]

cout << t.rangeQuery(1, 5).sum() << "\n"; // 14
```

### 线段树思想：特殊修改与结构变换

包含一些相对高频、但无法简单使用统一封装的特殊修改操作。这些操作无法在 $\mathcal O(1)$ 内由整段信息完成修改，因此不能直接套用上一节的统一封装（Info + Tag）。

虽然理论上依旧可以使用势能线段树做统一封装，但是实现起来会比较复杂，故在此单独列出。

#### 区间取模/取整开方、单点赋值、区间和

本质是势能剪枝，无法仅由区间和直接计算取模后的区间和。额外维护区间最大值：
- 取模时，若区间最大值小于模数，整段不变；两次赋值之间，单点真正取模至多 $\mathcal O(\log(A+1))$ 次。
- 开方时，若区间最大值不超过 $1$，整段不变；两次赋值之间，单点真正开方至多 $\mathcal O(\log\log(A+2))$ 次。

其中，$A$ 为初值及所有赋入值的最大值。要求所有元素始终非负，且每次模数 $v > 0$。

总均摊复杂度为 $\mathcal O((n+q)\log n\log(A+2))$。

```c++
template<class T> T isqrt(T n) { // n >= 0
    T x = sqrtl(n); // 如不需要兼容 `__int128`，可使用 `sqrt` 代替，常数略小一点
    while (x + 1 <= n / (x + 1)) ++x;
    while (x && x > n / x) --x;
    return x;
}

struct SegTreeShrink {
    int n;
    vector<i64> sum, mx;

    SegTreeShrink(const vector<i64> &a) : n(int(a.size()) - 1), sum(4 * n + 4), mx(4 * n + 4) {
        build(1, 1, n, a);
    }

    void pointSet(int x, i64 v) { pointSet(1, 1, n, x, v); }
    void rangeMod(int l, int r, i64 v) { rangeMod(1, 1, n, l, r, v); }
    void rangeSqrt(int l, int r) { rangeSqrt(1, 1, n, l, r); }
    i64 rangeSum(int l, int r) { return rangeSum(1, 1, n, l, r); }

private:
    void build(int p, int l, int r, const vector<i64> &a) {
        if (l == r) return sum[p] = mx[p] = a[l], void();
        int m = (l + r) / 2;
        build(p << 1, l, m, a);
        build(p << 1 | 1, m + 1, r, a);
        pull(p);
    }
    void pull(int p) {
        sum[p] = sum[p << 1] + sum[p << 1 | 1];
        mx[p] = max(mx[p << 1], mx[p << 1 | 1]);
    }
    void pointSet(int p, int l, int r, int x, i64 v) {
        if (l == r) return sum[p] = mx[p] = v, void();
        int m = (l + r) / 2;
        if (x <= m) pointSet(p << 1, l, m, x, v);
        else pointSet(p << 1 | 1, m + 1, r, x, v);
        pull(p);
    }
    void rangeMod(int p, int l, int r, int x, int y, i64 v) {
        if (mx[p] < v) return; // 整段取模后不变
        if (l == r) return sum[p] %= v, mx[p] = sum[p], void();
        int m = (l + r) / 2;
        if (x <= m) rangeMod(p << 1, l, m, x, y, v);
        if (m < y) rangeMod(p << 1 | 1, m + 1, r, x, y, v);
        pull(p);
    }
    void rangeSqrt(int p, int l, int r, int x, int y) {
        if (mx[p] <= 1) return; // 0、1 开方后不变
        if (l == r) return sum[p] = mx[p] = isqrt(sum[p]), void();
        int m = (l + r) / 2;
        if (x <= m) rangeSqrt(p << 1, l, m, x, y);
        if (m < y) rangeSqrt(p << 1 | 1, m + 1, r, x, y);
        pull(p);
    }
    i64 rangeSum(int p, int l, int r, int x, int y) {
        if (x <= l && r <= y) return sum[p];
        int m = (l + r) / 2;
        i64 ans = 0;
        if (x <= m) ans += rangeSum(p << 1, l, m, x, y);
        if (m < y) ans += rangeSum(p << 1 | 1, m + 1, r, x, y);
        return ans;
    }
};
```

#### 区间加、单点赋值、区间 gcd 查询

本质是维护差分前缀和，令 $d_1=a_1$，$d_i=a_i-a_{i-1}$。区间加 $l,r$ 加上 $v$ 时，只会改变两个位置：$d_l \to d_l+v$、$d_{r+1} \to d_{r+1}-v \left(r<n \right)$。由定义：

$$
\gcd(a_l,a_{l+1},\ldots,a_r)
=\gcd(a_l,d_{l+1},d_{l+2},\ldots,d_r)
$$

故树状数组维护差分前缀和，得到 $a_l$；线段树维护差分绝对值的 $\gcd$。

空间与初始化复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$。

```c++
struct RangeAddGcd {
    int n;
    vector<i64> d, bit, seg;

    RangeAddGcd(const vector<i64> &a) : n(int(a.size()) - 1), d(n + 1), bit(n + 1), seg(2 * n) {
        for (int i = 1; i <= n; i++) {
            d[i] = a[i] - (i == 1 ? 0 : a[i - 1]);
            bitAdd(i, d[i]);
            seg[n + i - 1] = gcd(d[i], 0LL);
        }
        for (int p = n - 1; p; p--) {
            seg[p] = gcd(seg[p << 1], seg[p << 1 | 1]);
        }
    }

    void rangeAdd(int l, int r, i64 v) { // a[l..r] += v
        changeD(l, v);
        changeD(r + 1, -v);
    }
    void pointSet(int x, i64 v) { // a[x] = v
        rangeAdd(x, x, v - pointValue(x));
    }
    i64 pointValue(int x) const { return prefixSum(x); }
    i64 rangeGcd(int l, int r) const {
        return gcd(pointValue(l), rangeGcdDiff(l + 1, r));
    }

private:
    void bitAdd(int x, i64 v) {
        for (; x <= n; x += x & -x) bit[x] += v;
    }
    i64 prefixSum(int x) const {
        i64 ans = 0;
        for (; x; x -= x & -x) ans += bit[x];
        return ans;
    }
    void changeD(int x, i64 v) { // d[x] += v
        if (x > n) return;
        d[x] += v;
        bitAdd(x, v);

        int p = n + x - 1;
        seg[p] = gcd(d[x], 0LL);
        for (p >>= 1; p; p >>= 1) {
            seg[p] = gcd(seg[p << 1], seg[p << 1 | 1]);
        }
    }
    i64 rangeGcdDiff(int l, int r) const {
        if (l > r) return 0;
        i64 ans = 0;
        for (l += n - 1, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) ans = gcd(ans, seg[l++]);
            if (r & 1) ans = gcd(ans, seg[--r]);
        }
        return ans;
    }
};
```

### 二维树状数组：矩形修改、矩形查询

空间与初始化复杂度均为 $\mathcal O(nm)$，单次操作复杂度 $\mathcal O(\log n\log m)$。

#### 单点加、矩形和

```c++
struct BIT_2D {
    int n, m;
    vector<vector<i64>> w;

    BIT_2D(int n, int m) : n(n), m(m), w(n + 1, vector<i64>(m + 1)) {}
    void pointAdd(int x, int y, i64 p) { // 单点 [x, y] 修改
        for (int i = x; i <= n; i += i & -i) {
            for (int j = y; j <= m; j += j & -j) {
                w[i][j] += p;
            }
        }
    }
    i64 pointSum(int x, int y) { return matrixSum(x, y, x, y); } // 单点 [x, y] 查询
    i64 matrixSum(int u, int v, int x, int y) { // 矩形和 [u, v] ~ [x, y] 查询
        auto sum = [&](int x, int y) {
            i64 ans = 0;
            for (int i = x; i; i -= i & -i) {
                for (int j = y; j; j -= j & -j) {
                    ans += w[i][j];
                }
            }
            return ans;
        };
        return sum(x, y) - sum(u - 1, y) - sum(x, v - 1) + sum(u - 1, v - 1);
    }
};
```

#### 矩形加、矩形和

常数比上两个版本略微大一些。

```c++
struct exBIT_2D {
    int n, m;
    vector<vector<i64>> b1, b2, b3, b4;

    exBIT_2D(int n, int m) : n(n), m(m),
        b1(n + 1, vector<i64>(m + 1)), b2(n + 1, vector<i64>(m + 1)),
        b3(n + 1, vector<i64>(m + 1)), b4(n + 1, vector<i64>(m + 1)) {}
    void pointAdd(int x, int y, i64 p) { matrixAdd(x, y, x, y, p); } // 单点 [x, y] 修改
    void matrixAdd(int u, int v, int x, int y, i64 p) { // 矩形 [u, v] ~ [x, y] 修改
        auto add = [&](int u, int v, i64 p) {
            for (int i = u; i <= n; i += i & -i) {
                for (int j = v; j <= m; j += j & -j) {
                    b1[i][j] += p;
                    b2[i][j] += p * (u - 1);
                    b3[i][j] += p * (v - 1);
                    b4[i][j] += p * (u - 1) * (v - 1);
                }
            }
        };
        x++, y++;
        add(u, v, p), add(x, v, -p);
        add(x, y, p), add(u, y, -p);
    }
    i64 pointSum(int x, int y) { return matrixSum(x, y, x, y); } // 单点 [x, y] 查询
    i64 matrixSum(int u, int v, int x, int y) { // 矩形和 [u, v] ~ [x, y] 查询
        auto ask = [&](int u, int v) {
            i64 ans = 0;
            for (int i = u; i; i -= i & -i) {
                for (int j = v; j; j -= j & -j) {
                    ans += b1[i][j] * u * v;
                    ans -= b2[i][j] * v;
                    ans -= b3[i][j] * u;
                    ans += b4[i][j];
                }
            }
            return ans;
        };
        return ask(x, y) - ask(u - 1, y) - ask(x, v - 1) + ask(u - 1, v - 1);
    }
};
```

### 坐标压缩与离散化

设收集到 $k$ 个值，初始化复杂度 $\mathcal O(k\log k)$，单次操作复杂度 $\mathcal O(\log k)$。

#### 简单实现

反查原值使用 `xs[p - 1]`。若 $x$ 未加入 `xs`，`rank(x)` 仅表示其应插入的位置，不能作为已有点的编号使用。

```c++
vector<i64> xs;
for (int i = 1; i <= n; i++) {
    xs.push_back(a[i]);
}
// 将所有后续修改、查询中可能出现的值也加入 xs

sort(xs.begin(), xs.end());
xs.erase(unique(xs.begin(), xs.end()), xs.end());

auto rank = [&](i64 x) { // x 已加入 xs，返回 1-based 编号
    return int(lower_bound(xs.begin(), xs.end(), x) - xs.begin()) + 1;
};
```

#### 封装

```c++
template<class T = i64> struct Compress {
    bool f = false; // 防止漏 Build
    vector<T> xs;

    void add(T x) {
        f = true;
        xs.push_back(x);
    }
    void build() {
        if (!f) return;
        sort(xs.begin(), xs.end());
        xs.erase(unique(xs.begin(), xs.end()), xs.end());
        f = false;
    }
    int size() {
        build();
        return int(xs.size());
    }
    int rank(T x) { // 要求 x 已加入 xs
        build();
        return int(lower_bound(xs.begin(), xs.end(), x) - xs.begin()) + 1;
    }
    T value(int p) { // 反查原值
        build();
        return xs[p - 1]; // 1 <= p <= size()
    }
};
```

### 树链剖分

标准重链剖分，按子树大小选择重儿子。空间与初始化复杂度 $\mathcal O(n)$；一条路径被拆为 $\mathcal O(\log n)$ 个 DFS 序区间。配合树状数组或线段树时，路径操作复杂度通常为 $\mathcal O(\log^2 n)$，子树操作仅对应一个区间。

将一条路径拆为若干 DFS 序区间。若区间结果可按任意顺序合并，则可直接接入（`BIT`、`LazySegTree`）；若答案依赖整个连续区间的全局分布（小波矩阵、动态开点权值线段树），或合并顺序有意义（字符串拼接、矩阵连乘），则不能直接套用。

#### 统一封装

结点 $x$ 子树内的点权区间为 `[in[x], out[x]]`，边权区间为 `[in[x] + 1, out[x]]`。`forEdgePath(x, y, f)` 用于处理这个差值，约定边 `(fa[v], v)` 的权值存于 `in[v]`，并跳过 LCA。

```c++
struct HLD {
    int n, cur = 0;
    vector<vector<int>> g;
    vector<int> fa, dep, siz, son, top, in, out;

    HLD(int n) : n(n), g(n + 1), fa(n + 1), dep(n + 1),
        siz(n + 1), son(n + 1), top(n + 1), in(n + 1), out(n + 1) {}

    void addEdge(int x, int y) {
        g[x].push_back(y);
        g[y].push_back(x);
    }
    void build(int root = 1) {
        cur = 0;
        dep[root] = 0;
        auto dfs1 = [&](auto &&self, int x, int p) -> void {
            fa[x] = p, siz[x] = 1, son[x] = 0;
            for (int y : g[x]) {
                if (y == p) continue;
                dep[y] = dep[x] + 1;
                self(self, y, x);
                siz[x] += siz[y];
                if (!son[x] || siz[y] > siz[son[x]]) son[x] = y;
            }
        };
        dfs1(dfs1, root, 0);
        auto dfs2 = [&](auto &&self, int x, int h) -> void {
            top[x] = h, in[x] = ++cur;
            if (son[x]) self(self, son[x], h);
            for (int y : g[x]) {
                if (y != fa[x] && y != son[x]) self(self, y, y);
            }
            out[x] = cur;
        };
        dfs2(dfs2, root, root);
    }
    int lca(int x, int y) const {
        while (top[x] != top[y]) {
            if (dep[top[x]] < dep[top[y]]) swap(x, y);
            x = fa[top[x]];
        }
        return dep[x] < dep[y] ? x : y;
    }
    pair<int, int> subtree(int x) const {
        return {in[x], out[x]};
    }

    template<class F> void splitPath(int x, int y, F &f, bool edge) const {
        while (top[x] != top[y]) {
            if (dep[top[x]] < dep[top[y]]) swap(x, y);
            f(in[top[x]], in[x]);
            x = fa[top[x]];
        }
        if (dep[x] > dep[y]) swap(x, y);
        int l = in[x] + (edge ? 1 : 0);
        if (l <= in[y]) f(l, in[y]);
    }
    template<class F> void forPath(int x, int y, F &&f) const { // 点权路径
        splitPath(x, y, f, false);
    }
    template<class F> void forEdgePath(int x, int y, F &&f) const { // 边权路径
        splitPath(x, y, f, true);
    }
};
```

### 小波矩阵：静态区间第 k 小

设不同值个数为 $\sigma$，初始化复杂度 $\mathcal O(n \log n)$，空间 $\mathcal O(n \log \sigma)$，单次操作复杂度 $\mathcal O(\log \sigma)$。有==压位版本==可优化空间到 $\mathcal O(n + \tfrac{n \log n}{64})$。注意事项：
- 查找第 $k$ 小时，需要自行保证第 $k$ 小存在（$1 \le k \le R - L + 1$）；

```c++
struct WaveletMatrix {
    int lg;
    vector<i64> val;
    vector<int> mid;
    vector<vector<int>> z; // 每层的 0 前缀和

    WaveletMatrix(const vector<i64> &a) {
        int n = (int)a.size() - 1;
        val.assign(a.begin() + 1, a.end());
        sort(val.begin(), val.end());
        val.erase(unique(val.begin(), val.end()), val.end());

        int m = val.size();
        lg = m == 1 ? 1 : __lg(m - 1) + 1;

        vector<int> cur(n + 1), nxt(n + 1);
        for (int i = 1; i <= n; i++) {
            cur[i] = lower_bound(val.begin(), val.end(), a[i]) - val.begin();
        }

        mid.resize(lg);
        z.assign(lg, vector<int>(n + 1));
        for (int d = 0; d < lg; d++) {
            int pos = lg - 1 - d;
            for (int i = 1; i <= n; i++) {
                z[d][i] = z[d][i - 1] + ((cur[i] >> pos & 1) == 0);
            }

            mid[d] = z[d][n];
            int p0 = 0, p1 = mid[d];
            for (int i = 1; i <= n; i++) {
                if (cur[i] >> pos & 1) nxt[++p1] = cur[i];
                else nxt[++p0] = cur[i];
            }
            cur.swap(nxt);
        }
    }

    i64 kth(int L, int R, int k) const {
        int ans = 0;
        for (int d = 0; d < lg; d++) {
            int zL = z[d][L - 1], zR = z[d][R];
            if (k <= zR - zL) {
                L = zL + 1, R = zR;
            } else {
                k -= zR - zL, ans |= 1 << (lg - 1 - d);
                L = mid[d] + L - zL, R = mid[d] + R - zR;
            }
        }
        return val[ans];
    }

    i64 kthLarge(int L, int R, int k) const {
        return kth(L, R, R - L + 2 - k);
    }
};
```

### 莫队：离线区间询问

本质是暴力的优化思想，离线重排使得左右端点只移动少量位置，让复杂度均摊到可控的范围内。当题目满足以下两个前提时，可以考虑使用：
- 区间端点移动时的状态变化，可以被拆分为维护单个位置的状态变化，且维护单个位置的时间复杂度可控（常数至 $\log n$ 级别较为常见）；
- 可以离线读取操作，后续操作的参数不依赖此前询问的答案。

函数主体、答案类型和回滚逻辑都随题目变化，不创建统一封装类。

#### 普通莫队

记 $q$ 次询问，块长 $B$，单个位置的修改为常数，空间复杂度 $\mathcal O(n + q)$，总时间复杂度 $\mathcal O(q \log q + qB + \tfrac{n^2}{B})$。通常情况下设 $n$ 与 $q$ 同阶，$B \approx \sqrt n$，总时间复杂度粗略可记 $\mathcal O(n \sqrt n)$。

块长的取值有多种选择，如 $q$ 远小于 $n$ 时，取 $B = \tfrac{n}{\sqrt q}$ 更合适。对应写法：`int B = max(1, int(n / sqrt((double)max(1, q))));`。

下面以询问区间不同颜色数为例，假设初始颜色在 $[1,n]$ 内。

```c++
struct Query {
    int l, r, id, k;
};

signed main() {
    int n, q;
    cin >> n >> q;
    vector<int> w(n + 1);
    for (int i = 1; i <= n; i++) cin >> w[i];
    
    vector<Query> query(q + 1);
    int B = max(1, int(sqrt(n))); // 计算块长
    for (int i = 1; i <= q; i++) {
        cin >> query[i].l >> query[i].r;
        query[i].id = i;
        query[i].k = (query[i].l - 1) / B; // 计算块号
    }

    // 按照块号和奇偶性排序
    sort(query.begin() + 1, query.end(), [&](const Query &x, const Query &y) {
        if (x.k != y.k) return x.k < y.k;
        return (x.k & 1) ? x.r < y.r : x.r > y.r;
    });
    
    int l = 1, r = 0, val = 0;
    vector<int> cnt(n + 1), ans(q + 1);
    auto add = [&](int x) {
        if (cnt[w[x]]++ == 0) ++val;
    };
    auto del = [&](int x) {
        if (--cnt[w[x]] == 0) --val;
    };
    for (int i = 1; i <= q; i++) {
        auto [ql, qr, id, k] = query[i];
        while (l > ql) add(--l);
        while (r < qr) add(++r);
        while (l < ql) del(l++);
        while (r > qr) del(r--);
        ans[id] = val;
    }
    for (int i = 1; i <= q; i++) cout << ans[i] << '\n';
}
```

核心函数即 `add` 和 `del` 函数，例如维护区间内各颜色出现次数的平方和 $\sum_c \textrm{cnt}_c^2$：

```c++
vector<i64> cnt(n + 1), ans(q + 1); // 同时将答案数组改为 i64

    val += 2 * cnt[w[x]] + 1;
    ++cnt[w[x]];
};

    val -= 2 * cnt[w[x]] - 1;
    --cnt[w[x]];
};
```

#### 带修莫队

把每个询问记录为 `(l,r,t,id)`，其中 $t$ 是该询问发生前已经执行的修改数。修改记录同时保存位置、修改前的值和修改后的值。排序时增加时间维度；移动时间时，正向应用修改，反向恢复旧值。

记 $q$ 次询问，块长 $B$，单个位置的修改为常数，空间复杂度 $\mathcal O(n + q)$。通常情况下设 $n$ 与 $q$ 同阶，$B \approx n^{\frac{2}{3}}$，总时间复杂度粗略可记 $\mathcal O(n^{\frac{5}{3}})$。

下面以询问区间不同颜色数为例，但额外加入了将 $a_p$ 改为 $x$ 的修改操作。假设初始颜色及所有修改后的颜色均在 $[1,n]$ 内。

```c++
struct Query {
    int l, r, t, id, kl, kr;
};
struct Change {
    int pos, pre, now;
};

signed main() {
    int n, q;
    cin >> n >> q;
    vector<int> w(n + 1);
    for (int i = 1; i <= n; i++) cin >> w[i];
    
    vector<Query> query(1);
    vector<Change> change(1);
    vector<int> work = w;
    int B = max(1, int(pow(n, 2.0 / 3)));
    for (int i = 1; i <= q; i++) {
        string type;
        cin >> type;
        if (type == "Query") {
            int l, r;
            cin >> l >> r;
            query.push_back({l, r, int(change.size()) - 1, int(query.size()), (l - 1) / B, (r - 1) / B});
        } else {
            int p, x;
            cin >> p >> x;
            change.push_back({p, work[p], x});
            work[p] = x;
        }
    }
    q = int(query.size()) - 1; // 更新询问数量

    // 按照块号和奇偶性排序
    sort(query.begin() + 1, query.end(), [](const Query &x, const Query &y) {
        if (x.kl != y.kl) return x.kl < y.kl;
        if (x.kr != y.kr) return (x.kl & 1) ? x.kr < y.kr : x.kr > y.kr;
        return (x.kr & 1) ? x.t < y.t : x.t > y.t;
    });
    
    int l = 1, r = 0, val = 0; 
    int t = 0; // 累计修改次数
    vector<int> cnt(n + 1), ans(q + 1);
    auto add = [&](int x) {
        if (cnt[w[x]]++ == 0) ++val;
    };
    auto del = [&](int x) {
        if (--cnt[w[x]] == 0) --val;
    };
    auto apply = [&](int x, bool f) { // 正向应用修改，反向恢复旧值
        const auto &c = change[x];
        int v = f ? c.now : c.pre;
        if (l <= c.pos && c.pos <= r) { // 修改位置在当前区间内，按顺序执行
            del(c.pos);
            w[c.pos] = v;
            add(c.pos);
        } else { // 不在区间内时只修改数组
            w[c.pos] = v;
        }
    };
    for (int i = 1; i <= q; i++) {
        auto [ql, qr, qt, id, kl, kr] = query[i];
        while (l > ql) add(--l);
        while (r < qr) add(++r);
        while (l < ql) del(l++);
        while (r > qr) del(r--);
        while (t < qt) apply(++t, true);
        while (t > qt) apply(t--, false);
        ans[id] = val;
    }
    for (int i = 1; i <= q; i++) cout << ans[i] << '\n';
}
```

### 主席树（可持久化线段树）

以 $\mathcal O(N\log N)$ 的时间复杂度建树、查询、修改。

```c++
struct PresidentTree {
    static constexpr int N = 2e5 + 10;
    int cntNodes, root[N];
    struct node {
        int l, r;
        int cnt;
    } tr[4 * N + 17 * N];
    void modify(int &u, int v, int l, int r, int x) {
        u = ++cntNodes;
        tr[u] = tr[v];
        tr[u].cnt++;
        if (l == r) return;
        int mid = (l + r) / 2;
        if (x <= mid)
            modify(tr[u].l, tr[v].l, l, mid, x);
        else
            modify(tr[u].r, tr[v].r, mid + 1, r, x);
    }
    int kth(int u, int v, int l, int r, int k) {
        if (l == r) return l;
        int res = tr[tr[v].l].cnt - tr[tr[u].l].cnt;
        int mid = (l + r) / 2;
        if (k <= res)
            return kth(tr[u].l, tr[v].l, l, mid, k);
        else
            return kth(tr[u].r, tr[v].r, mid + 1, r, k - res);
    }
};
```

### KD Tree

在第 $k$ 维上的单次查询复杂度最坏为 $\mathcal O(n^{1-k^{-1}})$。

```c++
struct KDT {
    constexpr static int N = 1e5 + 10, K = 2;
    double alpha = 0.725;
    struct node {
        int info[K];
        int mn[K], mx[K];
    } tr[N];
    int ls[N], rs[N], siz[N], id[N], d[N];
    int idx, rt, cur;
    int ans;
    KDT() {
        rt = 0;
        cur = 0;
        memset(ls, 0, sizeof ls);
        memset(rs, 0, sizeof rs);
        memset(d, 0, sizeof d);
    }
    void apply(int p, int son) {
        if (son) {
            for (int i = 0; i < K; i++) {
                tr[p].mn[i] = min(tr[p].mn[i], tr[son].mn[i]);
                tr[p].mx[i] = max(tr[p].mx[i], tr[son].mx[i]);
            }
            siz[p] += siz[son];
        }
    }
    void maintain(int p) {
        for (int i = 0; i < K; i++) {
            tr[p].mn[i] = tr[p].info[i];
            tr[p].mx[i] = tr[p].info[i];
        }
        siz[p] = 1;
        apply(p, ls[p]);
        apply(p, rs[p]);
    }
    int build(int l, int r) {
        if (l > r) return 0;
        vector<double> avg(K);
        for (int i = 0; i < K; i++) {
            for (int j = l; j <= r; j++) {
                avg[i] += tr[id[j]].info[i];
            }
            avg[i] /= (r - l + 1);
        }
        vector<double> var(K);
        for (int i = 0; i < K; i++) {
            for (int j = l; j <= r; j++) {
                var[i] += (tr[id[j]].info[i] - avg[i]) * (tr[id[j]].info[i] - avg[i]);
            }
        }
        int mid = (l + r) / 2;
        int x = max_element(var.begin(), var.end()) - var.begin();
        nth_element(id + l, id + mid, id + r + 1, [&](int a, int b) {
            return tr[a].info[x] < tr[b].info[x];
        });
        d[id[mid]] = x;
        ls[id[mid]] = build(l, mid - 1);
        rs[id[mid]] = build(mid + 1, r);
        maintain(id[mid]);
        return id[mid];
    }
    void print(int p) {
        if (!p) return;
        print(ls[p]);
        id[++idx] = p;
        print(rs[p]);
    }
    void rebuild(int &p) {
        idx = 0;
        print(p);
        p = build(1, idx);
    }
    bool bad(int p) {
        return alpha * siz[p] <= max(siz[ls[p]], siz[rs[p]]);
    }
    void insert(int &p, int cur) {
        if (!p) {
            p = cur;
            maintain(p);
            return;
        }
        if (tr[p].info[d[p]] > tr[cur].info[d[p]]) insert(ls[p], cur);
        else insert(rs[p], cur);
        maintain(p);
        if (bad(p)) rebuild(p);
    }
    void insert(vector<int> &a) {
        cur++;
        for (int i = 0; i < K; i++) {
            tr[cur].info[i] = a[i];
        }
        insert(rt, cur);
    }
    bool out(int p, vector<int> &a) {
        for (int i = 0; i < K; i++) {
            if (a[i] < tr[p].mn[i]) {
                return true;
            }
        }
        return false;
    }
    bool in(int p, vector<int> &a) {
        for (int i = 0; i < K; i++) {
            if (a[i] < tr[p].info[i]) {
                return false;
            }
        }
        return true;
    }
    bool all(int p, vector<int> &a) {
        for (int i = 0; i < K; i++) {
            if (a[i] < tr[p].mx[i]) {
                return false;
            }
        }
        return true;
    }
    void query(int p, vector<int> &a) {
        if (!p) return;
        if (out(p, a)) return;
        if (all(p, a)) {
            ans += siz[p];
            return;
        }
        if (in(p, a)) ans++;
        query(ls[p], a);
        query(rs[p], a);
    }
    int query(vector<int> &a) {
        ans = 0;
        query(rt, a);
        return ans;
    }
};
```

### 线性基

高效序列中子集的异或值。复杂度与数位相关，通常记为 $B = 63$，空间复杂度 $\mathcal O(B)$，单次操作复杂度 $\mathcal O(B)$；仅在线性基修改后首次查询第 $k$ 小值、合并两个线性基时，时间复杂度 $\mathcal O(B^2)$（重建）。

**默认子集非空**，若需要引入空集，可通过 `f=true` 控制；

#### 标准封装

动态维护一堆数字，没有顺序限制。注意事项：
- 数值范围为 $0 \le x < 2^B$；
- 查找第 $k$ 小时，注意 $1 \le k < 2^B$。

```c++
struct LB {
    static constexpr int B = 63;
    array<i64, B> d{};
    int base = 0; // 线性基的维数（基向量的数量）
    bool zero = false, dirty = false;

    bool insert(i64 x) { // 成功插入时返回 true
        for (int i = B - 1; i >= 0; i--) {
            if (!(x >> i & 1)) continue;
            if (d[i]) x ^= d[i];
            else {
                d[i] = x;
                base++;
                dirty = true;
                return true;
            }
        }
        zero = true; // 存在非空子集异或为 0
        return false;
    }
    bool contains(i64 x, bool f = false) const { // 是否存在一个子集能异或得到 x
        if (x == 0) return f || zero;
        for (int i = B - 1; i >= 0; i--) {
            if (!(x >> i & 1)) continue;
            if (!d[i]) return false;
            x ^= d[i];
        }
        return x == 0;
    }
    i64 maxXor(bool f = false) const { // 最大子集异或值
        if (!f && !base && !zero) return -1;
        return maxXorWith(0);
    }
    i64 maxXorWith(i64 x) const { // x 与任意子集异或后的最大值
        for (int i = B - 1; i >= 0; i--) {
            x = max(x, x ^ d[i]);
        }
        return x;
    }
    i64 minXor(bool f = false) const { // 最小子集异或值
        if (f || zero) return 0;
        for (int i = 0; i < B; i++) {
            if (d[i]) return d[i];
        }
        return -1;
    }
    void merge(const LB &b) { // 合并两个线性基
        zero |= b.zero; // 保留 b 中已有的非空零异或，重复元素按多次插入处理
        for (int i = B - 1; i >= 0; i--) {
            if (b.d[i]) insert(b.d[i]);
        }
    }
    i64 kth(i64 k, bool allowEmpty = false) { // 子集异或值去重后的第 k 小
        if (k <= 0) return -1;
        if (allowEmpty || zero) k--;
        if (k >> base) return -1; // 等价于 k >= 2^base
        if (dirty) { // 重建
            for (int i = 0; i < B; i++) {
                for (int j = i - 1; j >= 0; j--) {
                    if (d[i] >> j & 1) d[i] ^= d[j];
                }
            }
            dirty = false;
        }
        i64 ans = 0;
        for (int i = 0, j = 0; i < B; i++) {
            if (d[i]) {
                if (k >> j & 1) ans ^= d[i];
                j++;
            }
        }
        return ans;
    }
};
```

#### 序列任意连续区间导出

导出任意一个区间至普通线性基。空间与初始化复杂度 $\mathcal O(nB)$，单次操作复杂度 $\mathcal O(B)$。

对每一个前缀构建线性基实现，如果内存限制严格，考虑调整 $B$ 的大小，或采用离线做法：按右端点递增扫描，只保留当前前缀的基，降低至 $\mathcal O(B)$。

```c++
struct rLB {
    static constexpr int B = LB::B;
    vector<array<i64, B>> d;
    vector<array<int, B>> pos;

    rLB(const vector<i64> &a) : d(a.size()), pos(a.size()) { // a[1..n]
        for (int r = 1; r < int(a.size()); r++) {
            d[r] = d[r - 1];
            pos[r] = pos[r - 1];
            i64 x = a[r];
            int id = r;
            for (int i = B - 1; i >= 0; i--) {
                if (!(x >> i & 1)) continue;
                if (!d[r][i]) {
                    d[r][i] = x;
                    pos[r][i] = id;
                    break;
                }
                if (pos[r][i] < id) { // 同一最高位保留下标较大的向量
                    swap(d[r][i], x);
                    swap(pos[r][i], id);
                }
                x ^= d[r][i];
            }
        }
    }
    LB getLB(int l, int r) const { // 导出区间 l 到 r 的线性基
        LB ans;
        for (int i = 0; i < B; i++) {
            if (pos[r][i] >= l) {
                ans.d[i] = d[r][i]; // 最高位互异，可直接保留
                ans.base++;
            }
        }
        ans.zero = ans.base < r - l + 1;
        ans.dirty = true; // 尚未为 kth 重建
        return ans;
    }
};
```

#### 树上任意简单路径导出

导出任意一条简单路径至普通线性基。空间与初始化复杂度 $\mathcal O(nB)$，单次操作复杂度 $\mathcal O(B^2 + \log n)$。

```c++
struct tLB {
    static constexpr int B = LB::B;
    const HLD &hld;
    vector<array<i64, B>> d;
    vector<array<int, B>> pos;

    tLB(const HLD &hld, const vector<i64> &a) : hld(hld), d(hld.n + 1), pos(hld.n + 1) { // a[1..n]
        vector<int> seq(hld.n + 1);
        for (int x = 1; x <= hld.n; x++) seq[hld.in[x]] = x;
        for (int t = 1; t <= hld.n; t++) { // DFS 序保证父结点先构造
            int x = seq[t], p = hld.fa[x];
            d[x] = d[p];
            pos[x] = pos[p];
            i64 v = a[x];
            int id = hld.dep[x] + 1; // 深度加一，0 留作空位标记
            for (int i = B - 1; i >= 0; i--) {
                if (!(v >> i & 1)) continue;
                if (!d[x][i]) {
                    d[x][i] = v;
                    pos[x][i] = id;
                    break;
                }
                if (pos[x][i] < id) { // 同一最高位保留深度较大的向量
                    swap(d[x][i], v);
                    swap(pos[x][i], id);
                }
                v ^= d[x][i];
            }
        }
    }
    LB getLB(int x, int y) const { // 导出路径 x 到 y 的线性基
        int z = hld.lca(x, y), lim = hld.dep[z] + 1;
        LB ans;
        for (int i = 0; i < B; i++) { // z 到 x，包含 LCA
            if (pos[x][i] >= lim) {
                ans.d[i] = d[x][i]; // 最高位互异，可直接保留
                ans.base++;
            }
        }
        for (int i = B - 1; i >= 0; i--) { // z 到 y，排除 LCA
            if (pos[y][i] > lim) ans.insert(d[y][i]);
        }
        int len = hld.dep[x] + hld.dep[y] - 2 * hld.dep[z] + 1;
        ans.zero = ans.base < len; // 恢复整条路径的非空零异或信息
        ans.dirty = true; // 直接保留的基尚未为 kth 重建
        return ans;
    }
};
```

### 珂朵莉树：基于均摊与随机的区间操作

本质是暴力的优化思想，将值相同的一段区间合并成一个结点处理。当题目满足以下两个前提时，可以考虑使用：
- 区间**修改操作按段处理**而非按位处理，即相同的值经过同一个变换仍然相同，例如区间取模、异或、开方等操作；
- 需要**累计遍历的段数可控**，最典型的即含**区间赋值**操作，且范围与频率是**随机生成**的。

在满足以上条件的前提下，记初始 $n$ 段、执行 $q$ 次区间赋值操作，总复杂度 $\mathcal O((n+q)\log(n+q+1))$。混合其他修改和查询时，效率取决于遍历的段数，粗略可记 $\mathcal O(n \log\log n)$。

```c++
struct ODT {
    map<int, i64> s; // 段起点 -> 段值

    ODT(int n, i64 x = 0) : s{{1, x}, {n + 1, 0}} {} // a[1..n] 初始均为 x
    ODT(const vector<i64> &a) { // 按 a[1..n] 构造，合并初始相邻等值段
        int n = int(a.size()) - 1;
        for (int i = 1; i <= n; i++) {
            if (i == 1 || a[i] != a[i - 1]) s.emplace(i, a[i]);
        }
        s.emplace(n + 1, 0);
    }
    auto getRange(int l, int r) { // 对齐边界，返回供本次操作遍历的 [L, R)
        auto split = [&](int pos) { // 令 pos 成为段起点
            auto it = s.lower_bound(pos);
            if (it->first == pos) return it;
            return s.emplace_hint(it, pos, prev(it)->second);
        };
        return pair{split(l), split(r + 1)};
    }
    void rangeAssign(int l, int r, i64 x) { // 区间赋值，并将覆盖范围缩成一段
        auto [L, R] = getRange(l, r);
        s.erase(L, R);
        s.emplace(l, x);
    }
};
```

<blockquote>

典型应用为区间染色与颜色计数：区间染色后，维护各颜色的出现次数与全局不同颜色数。

```c++
int n = 1'000'000'000;
ODT t(n, 1);
map<i64, i64> cnt{{1, n}}; // 颜色 -> 出现次数
auto work = [&](int l, int r, i64 v) {
    auto [L, R] = t.getRange(l, r);
    for (auto it = L; it != R; ++it) {
        auto &num = cnt[it->second];
        num -= next(it)->first - it->first;
        if (num == 0) cnt.erase(it->second);
    }
    t.rangeAssign(l, r, v); // 被遍历的旧段随即删除，保证上述摊还复杂度
    cnt[v] += r - l + 1;
};
work(2, 4, 2);
work(3, 5, 3);
cout << cnt.size() << '\n'; // 全局不同颜色数：3
cout << cnt.at(2) << '\n'; // 颜色 2 的出现次数：1
cout << cnt.at(3) << '\n'; // 颜色 3 的出现次数：3
```

```c++
void rangeAdd(int l, int r, i64 x) { // 区间加
    auto [L, R] = getRange(l, r);
    for (auto it = L; it != R; ++it) it->second += x;
}

i64 kth(int l, int r, int k) { // 区间第 k 小
    assert(1 <= k && k <= r - l + 1);
    auto [L, R] = getRange(l, r);
    vector<pair<i64, int>> a;
    for (auto it = L; it != R; ++it) {
        a.emplace_back(it->second, next(it)->first - it->first);
    }
    sort(a.begin(), a.end());
    int p = 0;
    while (k > a[p].second) {
        k -= a[p].second;
        p++;
    }
    return a[p].first;
}
i64 rangePowerSum(int l, int r, int k, int mod) { // 区间幂次和
    assert(k >= 0 && mod > 0);
    auto power = [&](i64 x) {
        x = (x % mod + mod) % mod;
        i64 res = 1 % mod;
        for (int p = k; p; p >>= 1, x = x * x % mod) {
            if (p & 1) res = res * x % mod;
        }
        return res;
    };
    auto [L, R] = getRange(l, r);
    i64 ans = 0;
    for (auto it = L; it != R; ++it) {
        ans = (ans + power(it->second) * (next(it)->first - it->first)) % mod;
    }
    return ans;
}
```

### 快捷维护一个有序多重集合

动态地维护一个可重集合，支持常见六操作：插入、删除一个数、排名、第 $k$ 小、严格前驱、严格后继。

记录一些简易实现，注意事项：
- 删除元素时，需要自行保证目标元素存在；
- 查找前驱与后继时，应当自行保证严格前驱/后继存在；
- 查找第 $k$ 小时，需要自行保证第 $k$ 小存在。

#### 基于有序 vector

适合**规模较小或修改较少**的场景。

主要瓶颈是插入与删除操作，复杂度 $\mathcal O(n)$；二分查询、排名与前后继为 $\mathcal O(\log n)$，第 $k$ 小为 $\mathcal O(1)$。

```c++
struct VectorSet {
    vector<i64> a;

    VectorSet(vector<i64> &v) : a(v) {
        sort(a.begin(), a.end());
    }
    int size() { return a.size(); }
    void insert(i64 x) {
        a.insert(lower_bound(a.begin(), a.end(), x), x);
    }
    bool erase(i64 x) { // 删除第一个 x，返回是否成功
        auto it = lower_bound(a.begin(), a.end(), x);
        if (it == a.end() || *it != x) return false;
        a.erase(it);
        return true;
    }
    int rank(i64 x) {
        return lower_bound(a.begin(), a.end(), x) - a.begin() + 1;
    }
    i64 kth(int k) { return a[k - 1]; }
    i64 pre(i64 x) {
        return *prev(lower_bound(a.begin(), a.end(), x));
    }
    i64 suf(i64 x) {
        return *upper_bound(a.begin(), a.end(), x);
    }
    int count(i64 x) {
        auto [l, r] = equal_range(a.begin(), a.end(), x);
        return r - l;
    }
};
```

#### 基于 pbds

使用 `(值, 唯一编号)` 为键的技巧以支持重复值。空间复杂度 $\mathcal O(n)$，单次操作复杂度 $\mathcal O(\log n)$，但**常数较大**。

```c++
#include <ext/pb_ds/assoc_container.hpp>
#include <ext/pb_ds/tree_policy.hpp>
using namespace __gnu_pbds;

template<class T>
using ordered_set = tree<T, null_type, less<T>, rb_tree_tag, tree_order_statistics_node_update>;

struct PbdsSet {
    ordered_set<pair<i64, int>> s;
    int id = 0;

    int size() { return s.size(); }
    void insert(i64 x) { s.insert({x, ++id}); }
    bool erase(i64 x) { // 删除第一个 x，返回是否成功
        auto it = s.lower_bound({x, 0});
        if (it == s.end() || it->first != x) return false;
        s.erase(it);
        return true;
    }
    int rank(i64 x) { return s.order_of_key({x, 0}) + 1; }
    i64 kth(int k) { return s.find_by_order(k - 1)->first; }
    i64 pre(i64 x) {
        return prev(s.lower_bound({x, 0}))->first;
    }
    i64 suf(i64 x) {
        return s.lower_bound({x, id + 1})->first;
    }
    int count(i64 x) {
        return s.order_of_key({x, id + 1}) - s.order_of_key({x, 0});
    }
};
```

#### 基于树状数组

适用于**值域有界或可离线离散化**，值域过大且无法离线离散化时，使用平衡树替代。注意构造函数传入待维护值的闭区间 $[\text{lo}, \text{hi}]$，**所有操作值必须落在其中**。

记值域为 $U$，空间复杂度 $\mathcal O(U)$，单次操作复杂度 $\mathcal O(\log \text{U})$。

```c++
struct BITSet {
    int m, total = 0;
    i64 lo;
    vector<int> w;

    BITSet(i64 lo, i64 hi) : m(int(hi - lo + 1)), lo(lo), w(m + 1) {}
    int id(i64 x) { return int(x - lo + 1); } // 偏移
    void add(i64 x, int v) { // 单点 x 修改
        for (int p = id(x); p <= m; p += p & -p) {
            w[p] += v;
        }
    }
    int sum(int p) {
        int ans = 0;
        for (; p; p -= p & -p) ans += w[p];
        return ans;
    }

    int size() { return total; }
    void insert(i64 x) { add(x, 1), total++; }
    bool erase(i64 x) {
        if (!count(x)) return false;
        add(x, -1);
        total--;
        return true;
    }
    int rank(i64 x) { return sum(id(x) - 1) + 1; }
    i64 kth(int k) { // 要求频次非负，1 <= k <= size()
        int ans = 0;
        for (int i = 1 << __lg(m); i; i >>= 1) {
            int q = ans + i;
            if (q <= m && w[q] < k) {
                k -= w[q];
                ans = q;
            }
        }
        return lo + ans;
    }
    i64 pre(i64 x) { return kth(rank(x) - 1); }
    i64 suf(i64 x) { return kth(sum(id(x)) + 1); }
    int count(i64 x) { return sum(id(x)) - sum(id(x) - 1); }
};
```

### Treap：有序多重集合

采用 FHQ 的分裂合并方式，使用节点池，重复值分别建点，删除只断开节点，空间按累计建点数计算。记累计建点数为 $m$，空间复杂度 $\mathcal O(m)$，单次操作期望（受随机过程影响）复杂度 $\mathcal O(\log n)$。

#### 基本封装

```c++
struct FHQTreap {
    struct Node {
        int l = 0, r = 0, sz = 0, pri = 0;
        i64 val = 0;
    };
    int root = 0;
    vector<Node> t;
    mt19937 rng = mt19937(chrono::steady_clock::now().time_since_epoch().count());

    FHQTreap() : t(1) {}
    int newNode(i64 x) {
        Node u;
        u.sz = 1;
        u.pri = int(rng() >> 1);
        u.val = x;
        t.push_back(u);
        return int(t.size()) - 1;
    }
    void pull(int p) {
        t[p].sz = t[t[p].l].sz + t[t[p].r].sz + 1;
    }
    pair<int, int> split(int p, i64 x) { // < x，>= x
        if (!p) return {0, 0};
        if (t[p].val < x) {
            auto [a, b] = split(t[p].r, x);
            t[p].r = a;
            pull(p);
            return {p, b};
        } else {
            auto [a, b] = split(t[p].l, x);
            t[p].l = b;
            pull(p);
            return {a, p};
        }
    }
    int merge(int a, int b) { // 要求 a 的所有值 <= b 的所有值
        if (!a || !b) return a ? a : b;
        if (t[a].pri < t[b].pri) {
            t[a].r = merge(t[a].r, b);
            pull(a);
            return a;
        } else {
            t[b].l = merge(a, t[b].l);
            pull(b);
            return b;
        }
    }
    int eraseNode(int p, i64 x, bool &ok) { // 找到一个 x，合并其左右子树
        if (!p) return 0;
        if (t[p].val == x) {
            ok = true;
            return merge(t[p].l, t[p].r);
        }
        if (x < t[p].val) t[p].l = eraseNode(t[p].l, x, ok);
        else t[p].r = eraseNode(t[p].r, x, ok);
        pull(p);
        return p;
    }
    int sufCount(i64 x, bool f) const { // false: < x；true: <= x
        int ans = 0;
        for (int p = root; p; ) {
            if (t[p].val < x || (f && t[p].val == x)) {
                ans += t[t[p].l].sz + 1;
                p = t[p].r;
            } else {
                p = t[p].l;
            }
        }
        return ans;
    }

    int size() const { return t[root].sz; }
    void insert(i64 x) {
        int p = newNode(x);
        auto [a, b] = split(root, x);
        root = merge(merge(a, p), b);
    }
    bool erase(i64 x) {
        bool ok = false;
        root = eraseNode(root, x, ok);
        return ok;
    }
    int rank(i64 x) const { return sufCount(x, false) + 1; }
    i64 kth(int k) const { // 1 <= k <= size()
        int p = root;
        while (true) {
            int left = t[t[p].l].sz;
            if (k == left + 1) return t[p].val;
            if (k <= left) p = t[p].l;
            else {
                k -= left + 1;
                p = t[p].r;
            }
        }
    }
    i64 pre(i64 x) const { return kth(rank(x) - 1); }
    i64 suf(i64 x) const { return kth(sufCount(x, true) + 1); }
    int count(i64 x) const {
        return sufCount(x, true) - sufCount(x, false);
    }
};
```

#### 区间和、数值区间和、前 k 小之和

需要微调封装实现，本质上是对子树信息的求和。

```c++
struct Node { // 替换 Node
    int l = 0, r = 0, sz = 0, pri = 0;
    i64 val = 0, sum = 0;
};
int newNode(i64 x) { // 替换 newNode
    Node u;
    u.sz = 1;
    u.pri = int(rng() >> 1);
    u.val = u.sum = x;
    t.push_back(u);
    return int(t.size()) - 1;
}
void pull(int p) { // 替换 pull
    t[p].sz = t[t[p].l].sz + t[t[p].r].sz + 1;
    t[p].sum = t[t[p].l].sum + t[p].val + t[t[p].r].sum;
}

i64 kthSum(int k) const { // 增加函数，前 k 小之和，0 <= k <= size()
    i64 ans = 0;
    for (int p = root; p && k; ) {
        int left = t[p].l;
        if (k <= t[left].sz) {
            p = left;
        } else {
            ans += t[left].sum + t[p].val;
            k -= t[left].sz + 1;
            p = t[p].r;
        }
    }
    return ans;
}
i64 rangeSum(int l, int r) const { // 增加函数，第 l 小至第 r 小的元素和
    return kthSum(r) - kthSum(l - 1);
}
i64 rangeValueSum(i64 l, i64 r) const { // 增加函数，值落在 [l,r] 内的元素和
    auto sum = [&](i64 x, bool f) { // false: < x；true: <= x 的元素和
        i64 ans = 0;
        for (int p = root; p; ) {
            if (t[p].val < x || (f && t[p].val == x)) {
                ans += t[t[p].l].sum + t[p].val;
                p = t[p].r;
            } else {
                p = t[p].l;
            }
        }
        return ans;
    };
    return sum(r, true) - sum(l, false);
}
```

### 隐式 Treap：动态序列

整体封装策略与静态线段树、懒标记线段树一致。记累计建点数为 $m$，空间复杂度 $\mathcal O(m)$，初始化期望（受随机过程影响）复杂度 $\mathcal O(n \log n)$，导出序列 $\mathcal O(n)$，其余操作期望复杂度 $\mathcal O(\log n)$。

区间修改须能直接更新所维护的全部信息，且标记须与同段翻转可交换。更具体地，要求对同一段先整体修改再翻转，与先翻转再整体修改的结果相同。

#### 统一封装（Info + Tag）

适合序列插删、翻转、剪切拼接，同时维护区间信息。`Info()`、`operator+`、`Info::apply(v, len)`、`Tag::apply(v)` 沿用懒标记线段树的约定，标记按“旧操作后执行新操作”合成。需要额外定义：
- `Info::reverse()`，原地将信息更新为反向序列的信息；方向无关时函数体为空。

```c++
template<class Info, class Tag> struct FHQTreap {
    struct Node {
        int l = 0, r = 0, sz = 0, pri = 0;
        Info self, info; // 当前单点、整个子树的信息
        Tag tag;
        bool has = false, rev = false;
    };
    int root = 0;
    vector<Node> t;
    mt19937 rng = mt19937(chrono::steady_clock::now().time_since_epoch().count());

    FHQTreap() : t(1) {}
    explicit FHQTreap(const vector<Info> &a) : FHQTreap() {
        root = build(a);
    }
    int newNode(const Info &v) {
        Node u;
        u.sz = 1;
        u.pri = int(rng() >> 1);
        u.self = u.info = v;
        t.push_back(u);
        return int(t.size()) - 1;
    }
    int build(const vector<Info> &a) { // 用 a[1..n] 建一段，返回根，不修改 root
        int p = 0;
        for (int i = 1; i < int(a.size()); i++) {
            int q = newNode(a[i]);
            p = merge(p, q);
        }
        return p;
    }
    void pull(int p) {
        t[p].sz = t[t[p].l].sz + t[t[p].r].sz + 1;
        t[p].info = t[t[p].l].info + t[p].self + t[t[p].r].info;
    }
    void apply(int p, const Tag &v) {
        if (!p) return;
        t[p].self.apply(v, 1); // 单点与子树信息都要更新
        t[p].info.apply(v, t[p].sz);
        if (t[p].has) t[p].tag.apply(v);
        else t[p].tag = v, t[p].has = true;
    }
    void applyRev(int p) {
        if (!p) return;
        swap(t[p].l, t[p].r);
        t[p].info.reverse();
        t[p].rev ^= 1;
    }
    void push(int p) {
        if (t[p].rev) {
            applyRev(t[p].l);
            applyRev(t[p].r);
            t[p].rev = false;
        }
        if (t[p].has) {
            apply(t[p].l, t[p].tag);
            apply(t[p].r, t[p].tag);
            t[p].has = false;
        }
    }
    pair<int, int> split(int p, int k) { // 前 k 项，其余项；0 <= k <= t[p].sz
        if (!p) return {0, 0};
        push(p);
        int left = t[t[p].l].sz;
        if (k <= left) {
            auto [a, b] = split(t[p].l, k);
            t[p].l = b;
            pull(p);
            return {a, p};
        } else {
            auto [a, b] = split(t[p].r, k - left - 1);
            t[p].r = a;
            pull(p);
            return {p, b};
        }
    }
    int merge(int a, int b) { // 将 a 放在 b 前面；重连儿子前先下传
        if (!a || !b) return a ? a : b;
        if (t[a].pri < t[b].pri) {
            push(a);
            t[a].r = merge(t[a].r, b);
            pull(a);
            return a;
        } else {
            push(b);
            t[b].l = merge(a, t[b].l);
            pull(b);
            return b;
        }
    }
    int size() const { return t[root].sz; }
    void insert(int pos, const Info &v) { // 在 pos 前插入，1 <= pos <= size()+1
        int p = newNode(v);
        auto [a, b] = split(root, pos - 1);
        root = merge(merge(a, p), b);
    }
    void erase(int l, int r) { // 删除 [l,r]；删单点用 erase(pos, pos)
        auto [a, bc] = split(root, l - 1);
        auto [b, c] = split(bc, r - l + 1);
        root = merge(a, c);
    }
    void pointSet(int pos, const Info &v) {
        auto [a, bc] = split(root, pos - 1);
        auto [b, c] = split(bc, 1);
        t[b].self = t[b].info = v;
        t[b].has = t[b].rev = false; // 已切成单点，清除旧标记
        root = merge(merge(a, b), c);
    }
    Info pointGet(int pos) { return rangeQuery(pos, pos); }
    void reverse(int l, int r) {
        auto [a, bc] = split(root, l - 1);
        auto [b, c] = split(bc, r - l + 1);
        applyRev(b);
        root = merge(merge(a, b), c);
    }
    void rangeApply(int l, int r, const Tag &v) {
        auto [a, bc] = split(root, l - 1);
        auto [b, c] = split(bc, r - l + 1);
        apply(b, v);
        root = merge(merge(a, b), c);
    }
    Info rangeQuery(int l, int r) {
        auto [a, bc] = split(root, l - 1);
        auto [b, c] = split(bc, r - l + 1);
        Info ans = t[b].info;
        root = merge(merge(a, b), c);
        return ans;
    }
    vector<Info> getArray() { // 导出单点信息 a[1..size()]
        vector<Info> a(1);
        auto dfs = [&](auto &&self, int p) -> void {
            if (!p) return;
            push(p);
            self(self, t[p].l);
            a.push_back(t[p].self);
            self(self, t[p].r);
        };
        dfs(dfs, root);
        return a;
    }
};
```

#### 区间剪切拼接、区间循环移位

接入统一封装即可，操作复杂度保持不变。

```c++
void moveRange(int l, int r, int pos) { // 剪下 [l,r]，插到剩余序列的 pos 前
    // 1 <= pos <= size() - (r-l+1) + 1
    auto [a, bc] = split(root, l - 1);
    auto [b, c] = split(bc, r - l + 1);
    int rest = merge(a, c);
    auto [d, e] = split(rest, pos - 1);
    root = merge(merge(d, b), e);
}
void rotateRight(int l, int r, int k) { // [l,r] 循环右移 k，负数表示左移
    int len = r - l + 1;
    k %= len;
    if (k < 0) k += len;
    if (!k) return;
    auto [a, bc] = split(root, l - 1);
    auto [b, c] = split(bc, len);
    auto [d, e] = split(b, len - k);
    root = merge(merge(a, merge(e, d)), c);
}
```

#### 区间翻转、区间赋值、区间最大子段和

这一操作会改变序列的顺序，需要维护 `reverse` 标记。注意，其只与逐项修改（区间赋值）、重连合并（区间剪切拼接、区间循环移位）兼容，而**懒更新（区间加）会失效**。

沿用静态线段树的非空最大子段和信息，翻转时交换最大前后缀；赋值标记覆盖旧赋值。

```c++
struct Tag {
    i64 set = 0;
    void apply(const Tag &v) { set = v.set; }
};
struct Info {
    static constexpr i64 P = -(1LL << 60); // 必须小于所有可能的非空子段和
    i64 sum = 0;
    i64 lmx = P, rmx = P, mx = P;

    Info() = default;
    Info(i64 x) : sum(x), lmx(x), rmx(x), mx(x) {}
    Info(i64 s, i64 l, i64 r, i64 m) : sum(s), lmx(l), rmx(r), mx(m) {}
    void apply(const Tag &v, int len) {
        sum = v.set * len;
        lmx = rmx = mx = max(v.set, sum);
    }
    void reverse() { swap(lmx, rmx); }
};
Info operator+(const Info &a, const Info &b) {
    return {
        a.sum + b.sum,
        max(a.lmx, a.sum + b.lmx),
        max(b.rmx, b.sum + a.rmx),
        max({a.mx, b.mx, a.rmx + b.lmx})
    };
}

vector<Info> a = {{}, {-2}, {3}, {-1}, {4}, {-5}};
FHQTreap<Info, Tag> s(a);
cout << s.rangeQuery(1, 5).mx << '\n'; // 求第 1 到第 5 个位置的最大子段和：6
s.reverse(2, 4); // 翻转第 2 到第 4 个位置：-2 4 -1 3 -5
s.rangeApply(2, 4, {-2}); // 将第 2 到第 4 个位置赋值为 -2：-2 -2 -2 -2 -5
s.insert(3, {6}); // 在第 3 个位置前插入 6：-2 -2 6 -2 -2 -5
cout << s.rangeQuery(1, s.size()).mx << '\n'; // 求序列最大子段和：6
```


### 扫描线思想

将事件按一维坐标排序，扫描时用数据结构维护另一维的状态。

#### 二维偏序（逆序对统计）

统计 $i<j$ 且 $a_i>a_j$ 的点对，相等值按下标递增处理，不会重复贡献。

直接沿用树状数组封装 `BIT`，空间复杂度 $\mathcal O(n)$，总时间复杂度 $\mathcal O(n \log n)$。

```c++
i64 countInversions(const vector<i64> &a) {
    int n = int(a.size()) - 1;
    vector<pair<i64, int>> ord;
    for (int i = 1; i <= n; i++) ord.push_back({a[i], i});
    sort(ord.begin(), ord.end());

    BIT bit(n);
    i64 ans = 0;
    for (auto [v, x] : ord) {
        ans += bit.rangeSum(x + 1, n);
        bit.pointAdd(x, 1);
    }
    return ans;
}
```

#### 静态矩形区域稀疏点权和

离线查询闭矩形 $[x_1,x_2]\times[y_1,y_2]$ 内的点权和，允许点权为负。按横坐标扫描，压缩纵坐标，查询边界不必出现于点集。坐标范围较小且需要在线修改时，可使用二维树状数组。

直接沿用树状数组封装 `BIT`。记点数为 $n$、询问数为 $q$，空间复杂度 $\mathcal O(n+q)$，总时间复杂度 $\mathcal O((n+q)\log(n+q))$。

```c++
struct Point { i64 x, y, w; };

signed main() {
    int n, q;
    cin >> n >> q;
    vector<Point> points(n + 1);
    vector<i64> ys;
    for (int i = 1; i <= n; i++) {
        cin >> points[i].x >> points[i].y >> points[i].w;
        ys.push_back(points[i].y);
    }
    sort(ys.begin(), ys.end());
    ys.erase(unique(ys.begin(), ys.end()), ys.end());
    auto rank = [&](i64 x) {
        return lower_bound(ys.begin(), ys.end(), x) - ys.begin() + 1;
    };

    sort(points.begin() + 1, points.end(), [](const Point &a, const Point &b) {
        return a.x < b.x;
    });

    struct Event {
        i64 x;
        int lo, hi, id, sign; // sign：-1 表示左边界，1 表示右边界
        bool f; // 是否包含当前横坐标
    };
    vector<Event> l;
    for (int i = 1; i <= q; i++) {
        i64 x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        int lo = lower_bound(ys.begin(), ys.end(), y1) - ys.begin();
        int hi = upper_bound(ys.begin(), ys.end(), y2) - ys.begin();
        l.push_back({x1, lo, hi, i, -1, false}); // x < x1
        l.push_back({x2, lo, hi, i, 1, true});   // x <= x2
    }
    sort(l.begin(), l.end(), [](const Event &a, const Event &b) {
        if (a.x != b.x) return a.x < b.x;
        return a.f < b.f; // 同坐标先查 <，再查 <=
    });

    BIT bit(int(ys.size()));
    vector<i64> ans(q + 1);
    int cnt = 1;
    for (auto [x, lo, hi, id, sign, f] : l) {
        while (cnt <= n && (points[cnt].x < x || (f && points[cnt].x == x))) {
            bit.pointAdd(rank(points[cnt].y), points[cnt].w);
            ++cnt;
        }
        // 维护 lo 为小于 y1 的点坐标种数，hi 为不超过 y2 的种数
        ans[id] += sign * bit.rangeSum(lo + 1, hi);
    }

    for (int i = 1; i <= q; i++) cout << ans[i] << '\n';
}
```

#### 静态矩形面积并

查询若干个矩形的面积并，维护一棵临时线段树，用于恢复原始区间长度。空间复杂度 $\mathcal O(n)$，总时间复杂度 $\mathcal O(n\log n)$。注意事项：
- 矩形要求 $x_1 \lt x_2$，$y_1 \lt y_2$；
- 坐标差、长度与面积的中间运算均须在 `i64` 范围内。

```c++
struct SegTree {
    int n;
    vector<i64> ys, len;
    vector<int> cover; // 记录完整覆盖该节点的区间数，不是普通加法懒标记，无须向下传

    // ys 已排序去重，且至少有两个端点
    SegTree(const vector<i64> &ys)
        : n(int(ys.size()) - 1), ys(ys), len(4 * n + 4), cover(4 * n + 4) {}

    void rangeAdd(int l, int r, int v) { // 1 <= l <= r <= n；只能删除已加入的相同区间
        rangeAdd(1, 1, n, l, r, v);
    }
    i64 coveredLength() const { return len[1]; }

private:
    void pull(int p, int l, int r) {
        if (cover[p] > 0) len[p] = ys[r] - ys[l - 1];
        else if (l == r) len[p] = 0;
        else len[p] = len[p << 1] + len[p << 1 | 1];
    }
    void rangeAdd(int p, int l, int r, int x, int y, int v) {
        if (x <= l && r <= y) {
            cover[p] += v;
            pull(p, l, r);
            return;
        }
        int m = (l + r) / 2;
        if (x <= m) rangeAdd(p << 1, l, m, x, y, v);
        if (m < y) rangeAdd(p << 1 | 1, m + 1, r, x, y, v);
        pull(p, l, r);
    }
};

signed main() {
    int n;
    cin >> n;

    struct Event {
        i64 x, y1, y2;
        int v;
    };
    vector<Event> l;
    vector<i64> ys;
    for (int i = 1; i <= n; i++) {
        i64 x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        ys.push_back(y1);
        ys.push_back(y2);
        l.push_back({x1, y1, y2, 1});
        l.push_back({x2, y1, y2, -1});
    }
    sort(ys.begin(), ys.end());
    ys.erase(unique(ys.begin(), ys.end()), ys.end());
    auto rank = [&](i64 x) {
        return lower_bound(ys.begin(), ys.end(), x) - ys.begin() + 1;
    };

    sort(l.begin(), l.end(), [](const Event &a, const Event &b) {
        return a.x < b.x;
    });

    SegTree seg(ys);
    i64 ans = 0, lastX = l[0].x;
    for (auto [x, y1, y2, v] : l) {
        ans += seg.coveredLength() * (x - lastX); // 先结算上一条带
        seg.rangeAdd(rank(y1), rank(y2) - 1, v);
        lastX = x;
    }
    cout << ans << '\n';
}
```

### 常见结论

题意：（区间移位问题）要求将整个序列左移/右移若干个位置，例如，原序列为 $A=(a_1, a_2, \dots, a_n)$ ，右移 $x$ 位后变为 $A=(a_{x+1}, a_{x+2}, \dots, a_n,a_1,a_2,\dots, a_x)$ 。

区间的端点只是一个数字，即使被改变了，通过一定的转换也能够还原，所以我们可以 $\mathcal O(1)$ 解决这一问题。为了方便计算，我们规定下标从 $0$ 开始，即整个线段的区间为 $[0, n)$ ，随后，使用一个偏移量 `shift` 记录。使用 `shift = (shift + x) % n;` 更新偏移量；此后的区间查询/修改前，再将坐标偏移回去即可，下方代码使用区间修改作为示例。

```c++
cin >> l >> r >> x;
l--; // 坐标修改为 0 开始
r--;
l = (l + shift) % n; // 偏移
r = (r + shift) % n;
if (l > r) { // 区间分离则分别操作
    segt.modify(l, n - 1, x);
    segt.modify(0, r, x);
} else {
    segt.modify(l, r, x);
}
```

### 常见例题

题意：（带修莫队 - 维护队列）要求能够处理以下操作：

- `'Q' l r` ：询问区间 $[l,r]$ 有几个颜色；
- `'R' idx w` ：将下标 $\tt idx$ 的颜色修改为 $\tt w$ 。

输入格式为：第一行 $n$ 和 $q\ (1\le n, q\le 133333)$ 分别代表区间长度和操作数量；第二行 $n$ 个整数 $a_1,a_2\dots,a_n\ (1\le a_i\le 10^6)$ 代表初始颜色；随后 $q$ 行为具体操作。

```c++
const int N = 1e6 + 7;
signed main() {
    int n, q;
    cin >> n >> q;
    vector<int> w(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> w[i];
    }
    
    vector<array<int, 4>> query = {{}}; // {左区间, 右区间, 累计修改次数, 下标}
    vector<array<int, 2>> modify = {{}}; // {修改的值, 修改的元素下标}
    for (int i = 1; i <= q; i++) {
        char op;
        cin >> op;
        if (op == 'Q') {
            int l, r;
            cin >> l >> r;
            query.push_back({l, r, (int)modify.size() - 1, (int)query.size()});
        } else {
            int idx, w;
            cin >> idx >> w;
            modify.push_back({w, idx});
        }
    }
    
    int Knum = 2154; // 计算块长
    vector<int> K(n + 1);
    for (int i = 1; i <= n; i++) { // 固定块长
        K[i] = (i - 1) / Knum + 1;
    }
    sort(query.begin() + 1, query.end(), [&](auto x, auto y) {
        if (K[x[0]] != K[y[0]]) return x[0] < y[0];
        if (K[x[1]] != K[y[1]]) return x[1] < y[1];
        return x[3] < y[3];
    });
    
    int l = 1, r = 0, val = 0;
    int t = 0; // 累计修改次数
    vector<int> ans(query.size()), cnt(N);
    for (int i = 1; i < query.size(); i++) {
        auto [ql, qr, qt, id] = query[i];
        auto add = [&](int x) -> void {
            if (cnt[x] == 0) ++ val;
            ++ cnt[x];
        };
        auto del = [&](int x) -> void {
            -- cnt[x];
            if (cnt[x] == 0) -- val;
        };
        auto time = [&](int x, int l, int r) -> void {
            if (l <= modify[x][1] && modify[x][1] <= r) { //当修改的位置在询问期间内部时才会改变num的值
                del(w[modify[x][1]]);
                add(modify[x][0]);
            }
            swap(w[modify[x][1]], modify[x][0]); //直接交换修改数组的值与原始值，减少额外的数组开销，且方便复原
        };
        while (l > ql) add(w[--l]);
        while (r < qr) add(w[++r]);
        while (l < ql) del(w[l++]);
        while (r > qr) del(w[r--]);
        while (t < qt) time(++t, ql, qr);
        while (t > qt) time(t--, ql, qr);
        ans[id] = val;
    }
    for (int i = 1; i < ans.size(); i++) {
        cout << ans[i] << endl;
    }
}
```
