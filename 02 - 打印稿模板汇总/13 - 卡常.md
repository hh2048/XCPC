## 第十三章：卡常

### 数论

#### 最大公约数 gcd | Stein 算法

相较欧几里得算法，通过位移和减法替代传统除法运算，常数更小，达到 $\mathcal O(\log \max\{a, b\})$。本机测试 GNU `std::gcd` 同为 Stein 算法，常数以评测机为准。

```c++
i64 gcd(i64 a, i64 b) { // a, b >= 0
    if (!a || !b) return a | b;
    int t = __builtin_ctzll(a | b);
    a >>= __builtin_ctzll(a);
    while (b) {
        b >>= __builtin_ctzll(b);
        if (a > b) swap(a, b);
        b -= a;
    }
    return a << t;
}
```

适配第十四章 `BigInt`，设最大块数为 $n$，最坏复杂度 $\mathcal O(n^2)$。以此替换普通 `BigInt gcd`，二者不可同时复制；适用于频繁高精约分。

```c++
BigInt gcd(BigInt a, BigInt b) {
    a.s = b.s = 1;
    if (a.a.empty() || b.a.empty()) return a + b;
    int t = 0;
    while (!(a % 2) && !(b % 2)) a /= 2, b /= 2, ++t;
    while (!(a % 2)) a /= 2;
    do {
        while (!(b % 2)) b /= 2;
        if (a > b) swap(a, b);
        b -= a;
    } while (!b.a.empty());
    while (t--) a *= 2;
    return a;
}
```

#### 朴素查询：试除法 | 预分类讨论加速

约三倍常数优化，达到 $\mathcal O(\tfrac {\sqrt n}{3})$。

```c++
bool is_prime(int n) {
    if (n < 2) return false;
    if (n == 2 || n == 3) return true;
    if (n % 6 != 1 && n % 6 != 5) return false;
    for (int i = 5, j = n / i; i <= j; i += 6) {
        if (n % i == 0 || n % (i + 2) == 0) {
            return false;
        }
    }
    return true;
}
```

#### 单个大数：Pollard-Rho | Montgomery 幂模

记 $p$ 为当前合数的最小质因子，Pollard-Rho 单次查询期望复杂度 $\mathcal O(\sqrt{p}) \approx \mathcal O(n^{1/4})$，由于采取随机化算法，实践中通常跑得较快。

引入 Montgomery 幂模优化乘法取模的常数。注意事项：
- 需要满足 $0 \lt n \lt 2^{63}$；
- 需要满足模数为奇数，且 $1 \lt \text{MOD} \lt 2^{63}$。

```c++
using u64 = uint64_t;
using u128 = __uint128_t;

struct Montgomery64 {
    u64 m, im, u, v;
    Montgomery64(u64 p) : m(p) {
        im = p;
        for (int i = 0; i < 5; ++i) im *= 2 - p * im;
        im = -im;
        u = -p % p;
        v = u64(-u128(p) % p);
    }
    u64 reduce(u128 x) const {
        u64 q = u64(x) * im, y = u64((x + u128(q) * m) >> 64);
        return y < m ? y : y - m;
    }
    u64 trans(u64 x) const { return reduce(u128(x) * v); } // x < m
    u64 mul(u64 a, u64 b) const { return reduce(u128(a) * trans(b)); } // a,b < m
    u64 pow(u64 a, u64 n) const {
        u64 r = u, b = trans(a % m);
        for (; n; n >>= 1, b = reduce(u128(b) * b)) {
            if (n & 1) r = reduce(u128(r) * b);
        }
        return reduce(r);
    }
};

bool isprime(i64 n) { // Miller-Rabin
    if (n < 2 || n % 6 % 4 != 1) return (n | 1) == 3;
    u64 m = n, s = __builtin_ctzll(m - 1), d = m >> s;
    Montgomery64 M(m);
    for (u64 a : {2, 325, 9'375, 28'178, 450'775, 9'780'504, 1'795'265'022}) {
        if (a % m == 0) continue;
        u64 p = M.pow(a, d);
        if (p == 1 || p == m - 1) continue;
        for (u64 i = 1; i < s && p != m - 1; ++i) p = M.mul(p, p);
        if (p != m - 1) return false;
    }
    return true;
}

u64 rho(u64 n) {
    static mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
    if (!(n & 1)) return 2;
    Montgomery64 M(n);
    while (true) {
        u64 c = rng() % (n - 1) + 1;
        u64 y = rng() % n;
        u64 q = 1, g = 1, x = y, ys = y;
        auto f = [&](u64 x) {
            u64 z = M.mul(x, x) + c;
            return z < n ? z : z - n;
        };
        for (u64 r = 1; g == 1; r <<= 1) {
            x = y;
            for (u64 i = 0; i < r; ++i) y = f(y);
            q = 1;
            for (u64 k = 0; k < r && g == 1; k += 64) {
                ys = y;
                for (u64 i = 0, lim = min<u64>(64, r - k); i < lim; ++i) {
                    y = f(y), q = M.mul(q, x > y ? x - y : y - x);
                }
                g = gcd(q, n);
            }
        }
        if (g == n) do {
            ys = f(ys), g = gcd(x > ys ? x - ys : ys - x, n);
        } while (g == 1);
        if (g < n) return g;
    }
}

vector<i64> factorize(i64 x) { // Pollard-Rho
    vector<i64> res;
    auto dfs = [&](auto self, u64 n) -> void {
        if (n == 1) return;
        if (isprime(i64(n))) {
            res.push_back(i64(n));
            return;
        }
        u64 d = rho(n);
        self(self, d), self(self, n / d);
    };
    dfs(dfs, x);
    sort(res.begin(), res.end());
    return res;
}
```

### 网络流

#### 最大流 | 预流推进 HLPP

理论最坏复杂度为 $\mathcal O(N^2\sqrt M)$ ，例题范围：$N=1200,\ m=1.2\times 10^5$ 。

```c++
template <typename T> struct PushRelabel {
    const int inf = 0x3f3f3f3f;
    const T INF = 0x3f3f3f3f3f3f3f3f;
    struct Edge {
        int to, cap, flow, anti;
        Edge(int v = 0, int w = 0, int id = 0) : to(v), cap(w), flow(0), anti(id) {}
    };
    vector<vector<Edge>> e; 
    vector<vector<int>> gap;
    vector<T> ex; // 超额流
    vector<bool> ingap;
    vector<int> h;
    int n, gobalcnt, maxH = 0;
    T maxflow = 0;

    PushRelabel(int n) : n(n), e(n + 1), ex(n + 1), gap(n + 1) {}
    void addedge(int u, int v, int w) {
        e[u].push_back({v, w, (int)e[v].size()});
        e[v].push_back({u, 0, (int)e[u].size() - 1});
    }
    void PushEdge(int u, Edge &edge) {
        int v = edge.to, d = min(ex[u], 1LL * edge.cap - edge.flow);
        ex[u] -= d;
        ex[v] += d;
        edge.flow += d;
        e[v][edge.anti].flow -= d;
        if (h[v] != inf && d > 0 && ex[v] == d && !ingap[v]) {
            ++gobalcnt;
            gap[h[v]].push_back(v);
            ingap[v] = 1;
        }
    }
    void PushPoint(int u) {
        for (auto k = e[u].begin(); k != e[u].end(); k++) {
            if (h[k->to] + 1 == h[u] && k->cap > k->flow) {
                PushEdge(u, *k);
                if (!ex[u]) break;
            }
        }
        if (!ex[u]) return;
        if (gap[h[u]].empty()) {
            for (int i = h[u] + 1; i <= min(maxH, n); i++) {
                for (auto v : gap[i]) {
                    ingap[v] = 0;
                }
                gap[i].clear();
            }
        }
        h[u] = inf;
        for (auto [to, cap, flow, anti] : e[u]) {
            if (cap > flow) {
                h[u] = min(h[u], h[to] + 1);
            }
        }
        if (h[u] >= n) return;
        maxH = max(maxH, h[u]);
        if (!ingap[u]) {
            gap[h[u]].push_back(u);
            ingap[u] = 1;
        }
    }
    void init(int t, bool f = 1) {
        ingap.assign(n + 1, 0);
        for (int i = 1; i <= maxH; i++) {
            gap[i].clear();
        }
        gobalcnt = 0, maxH = 0;
        queue<int> q;
        h.assign(n + 1, inf);
        h[t] = 0, q.push(t);
        while (q.size()) {
            int u = q.front();
            q.pop(), maxH = h[u];
            for (auto &[v, cap, flow, anti] : e[u]) {
                if (h[v] == inf && e[v][anti].cap > e[v][anti].flow) {
                    h[v] = h[u] + 1;
                    q.push(v);
                    if (f) {
                        gap[h[v]].push_back(v);
                        ingap[v] = 1;
                    }
                }
            }
        }
    }
    T work(int s, int t) {
        init(t, 0);
        if (h[s] == inf) return maxflow;
        h[s] = n;
        ex[s] = INF;
        ex[t] = -INF;
        for (auto k = e[s].begin(); k != e[s].end(); k++) {
            PushEdge(s, *k);
        }
        while (maxH > 0) {
            if (gap[maxH].empty()) {
                maxH--;
                continue;
            }
            int u = gap[maxH].back();
            gap[maxH].pop_back();
            ingap[u] = 0;
            PushPoint(u);
            if (gobalcnt >= 10 * n) {
                init(t);
            }
        }
        ex[s] -= INF;
        ex[t] += INF;
        return maxflow = ex[t];
    }
};
```

### 数据结构

#### RMQ 问题 | 状压优化

严格 $\mathcal O(n)$ 预处理，$\mathcal O(1)$ 查询。

```c++
template<class T, class Cmp = less<T>> struct RMQ {
    const Cmp cmp = Cmp();
    static constexpr unsigned B = 64;
    using u64 = unsigned long long;
    int n;
    vector<vector<T>> a;
    vector<T> pre, suf, ini;
    vector<u64> stk;
    RMQ() {}
    RMQ(const vector<T> &v) {
        init(v);
    }
    void init(const vector<T> &v) {
        n = v.size();
        pre = suf = ini = v;
        stk.resize(n);
        if (!n) {
            return;
        }
        const int M = (n - 1) / B + 1;
        const int lg = __lg(M);
        a.assign(lg + 1, vector<T>(M));
        for (int i = 0; i < M; i++) {
            a[0][i] = v[i * B];
            for (int j = 1; j < B && i * B + j < n; j++) {
                a[0][i] = min(a[0][i], v[i * B + j], cmp);
            }
        }
        for (int i = 1; i < n; i++) {
            if (i % B) {
                pre[i] = min(pre[i], pre[i - 1], cmp);
            }
        }
        for (int i = n - 2; i >= 0; i--) {
            if (i % B != B - 1) {
                suf[i] = min(suf[i], suf[i + 1], cmp);
            }
        }
        for (int j = 0; j < lg; j++) {
            for (int i = 0; i + (2 << j) <= M; i++) {
                a[j + 1][i] = min(a[j][i], a[j][i + (1 << j)], cmp);
            }
        }
        for (int i = 0; i < M; i++) {
            const int l = i * B;
            const int r = min(1U * n, l + B);
            u64 s = 0;
            for (int j = l; j < r; j++) {
                while (s && cmp(v[j], v[__lg(s) + l])) {
                    s ^= 1ULL << __lg(s);
                }
                s |= 1ULL << (j - l);
                stk[j] = s;
            }
        }
    }
    T operator()(int l, int r) {
        if (l / B != (r - 1) / B) {
            T ans = min(suf[l], pre[r - 1], cmp);
            l = l / B + 1;
            r = r / B;
            if (l < r) {
                int k = __lg(r - l);
                ans = min({ans, a[k][l], a[k][r - (1 << k)]}, cmp);
            }
            return ans;
        } else {
            int x = B * (l / B);
            return ini[__builtin_ctzll(stk[r - 1] >> (l - x)) + l];
        }
    }
};
```

#### 小波矩阵 | 位向量压位

设不同值个数为 $\sigma$，初始化复杂度 $\mathcal O(n \log n)$，==空间优化至== $\mathcal O(n + \tfrac{n \log n}{64})$，单次操作复杂度 $\mathcal O(\log \sigma)$。

```c++
struct WaveletMatrixPacked {
    struct Bit {
        vector<u64> b;
        vector<int> pre;

        Bit(int n) : b((n + 63) >> 6), pre(b.size() + 1) {}

        int rank(int x) const { // [0, x) 内 1 的个数
            int p = x >> 6, off = x & 63;
            return pre[p] + (off ? __builtin_popcountll(b[p] & ((1ULL << off) - 1)) : 0);
        }
    };

    int lg;
    vector<i64> val;
    vector<int> mid;
    vector<Bit> bv;

    WaveletMatrixPacked(const vector<i64> &a) {
        int n = (int)a.size() - 1;
        vector<int> cur(n + 1), nxt(n + 1);

        {
            vector<pair<i64, int>> ord(n);
            for (int i = 1; i <= n; i++) ord[i - 1] = {a[i], i};
            sort(ord.begin(), ord.end());

            val.reserve(n);
            for (int i = 0; i < n; i++) {
                if (!i || ord[i].first != ord[i - 1].first) val.push_back(ord[i].first);
                cur[ord[i].second] = (int)val.size() - 1;
            }
        }

        int m = val.size();
        lg = m == 1 ? 1 : __lg(m - 1) + 1;

        mid.resize(lg);
        bv.reserve(lg);

        for (int d = 0; d < lg; d++) {
            int pos = lg - 1 - d;
            bv.emplace_back(n);
            auto &now = bv.back();

            for (int i = 1; i <= n; i++) {
                if (cur[i] >> pos & 1) now.b[i >> 6] |= 1ULL << (i & 63);
            }
            for (int i = 0; i < (int)now.b.size(); i++) {
                now.pre[i + 1] = now.pre[i] + __builtin_popcountll(now.b[i]);
            }

            mid[d] = n - now.rank(n);
            int p0 = 0, p1 = mid[d];
            for (int i = 1; i <= n; i++) {
                if (cur[i] >> pos & 1) nxt[++p1] = cur[i];
                else nxt[++p0] = cur[i];
            }
            cur.swap(nxt);
        }
    }

    i64 kth(int L, int R, int k) const {
        int ans = 0;
        for (int d = 0; d < lg; d++) {
            int oneL = bv[d].rank(L - 1), oneR = bv[d].rank(R);
            int cnt0 = R - L + 1 - (oneR - oneL);

            if (k <= cnt0) {
                L -= oneL, R -= oneR;
            } else {
                k -= cnt0, ans |= 1 << (lg - 1 - d);
                L = mid[d] + oneL + 1, R = mid[d] + oneR;
            }
        }
        return val[ans];
    }

    i64 kthLarge(int L, int R, int k) const {
        return kth(L, R, R - L + 2 - k);
    }
};
```

### 杂类

#### 快读

注意读入到文件结尾才结束，直接运行会无输出。

```c++
char buf[1 << 21], *p1 = buf, *p2 = buf;
inline char getc() {
    return p1 == p2 && (p2 = (p1 = buf) + fread(buf, 1, 1 << 21, stdin), p1 == p2) ? 0 : *p1++;
}
template<typename T> void Cin(T &a) {
    T ans = 0;
    bool f = 0;
    char c = getc();
    for (; c < '0' || c > '9'; c = getc()) {
        if (c == '-') f = -1;
    }
    for (; c >= '0' && c <= '9'; c = getc()) {
        ans = ans * 10 + c - '0';
    }
    a = f ? -ans : ans;
}
template<typename T, typename... Args> void Cin(T &a, Args &...args) {
    Cin(a), Cin(args...);
}
template<typename T> void Cout(T x) { // 注意，这里输出不带换行
    if (x < 0) putchar('-'), x = -x;
    if (x > 9) Cout(x / 10);
    putchar(x % 10 + '0');
}
```

#### 高精度整数类 | Karatsuba 大数乘法

设块数为 $n=\tfrac{|x|}{W}$，$m=\tfrac{|y|}{W}$，单次操作复杂度 $\mathcal O(\max\{n, m\}^{\log_2 3}) \approx \mathcal O(\max\{n, m\}^{1.58496})$。这一算法会将数位从朴素的 $10^9$ 进制转化为 $10^6$ 进制（防止 `i64` 溢出），仅适合两个数均很长且位数同阶时使用，并非朴素算法完全上位替代。

```c++
using v64 = vector<i64>;
vector<int> changeBase(const vector<int> &a, int oldW, int newW) {
    vector<i64> pw(max(oldW, newW) + 1, 1);
    for (int i = 1; i < int(pw.size()); i++) pw[i] = pw[i - 1] * 10;
    vector<int> b;
    i64 cur = 0;
    int w = 0;
    for (int x : a) {
        cur += x * pw[w], w += oldW;
        while (w >= newW) {
            b.push_back(cur % pw[newW]);
            cur /= pw[newW];
            w -= newW;
        }
    }
    if (cur) b.push_back(cur);
    return b;
}
v64 karatsuba(const v64 &a, const v64 &b) {
    int n = a.size();
    v64 c(n + n);
    if (n <= 32) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; ++j) {
                c[i + j] += a[i] * b[j];
            }
        }
        return c;
    }
    int k = n / 2;
    v64 a0(a.begin(), a.begin() + k), a1(a.begin() + k, a.end());
    v64 b0(b.begin(), b.begin() + k), b1(b.begin() + k, b.end());
    v64 c0 = karatsuba(a0, b0), c2 = karatsuba(a1, b1);
    for (int i = 0; i < k; i++) a1[i] += a0[i], b1[i] += b0[i];
    v64 c1 = karatsuba(a1, b1);
    for (int i = 0; i < int(c0.size()); i++) c1[i] -= c0[i];
    for (int i = 0; i < int(c2.size()); i++) c1[i] -= c2[i];
    for (int i = 0; i < int(c0.size()); i++) c[i] += c0[i];
    for (int i = 0; i < int(c1.size()); i++) c[i + k] += c1[i];
    for (int i = 0; i < int(c2.size()); i++) c[i + n] += c2[i];
    return c;
}
BigInt mulFast(const BigInt &x, const BigInt &y) {
    if (x.a.empty() || y.a.empty()) return {};
    vector<int> xa = changeBase(x.a, BigInt::W, 6);
    vector<int> ya = changeBase(y.a, BigInt::W, 6);
    int n = 1;
    while (n < int(max(xa.size(), ya.size()))) n <<= 1;
    v64 a(xa.begin(), xa.end()), b(ya.begin(), ya.end());
    a.resize(n), b.resize(n);
    v64 c = karatsuba(a, b);
    BigInt z;
    z.s = x.s * y.s;
    for (i64 carry = 0, i = 0; i < int(c.size()) || carry; i++) {
        i64 cur = carry + (i < int(c.size()) ? c[i] : 0);
        z.a.push_back(cur % 1'000'000);
        carry = cur / 1'000'000;
    }
    z.a = changeBase(z.a, 6, BigInt::W);
    z.trim();
    return z;
}
```

可重载乘号以直接嵌套适配其他封装，如分数运算类 `Frac`。但是会完全替代朴素高精乘法，两者不能共存：

```c++
BigInt operator*(const BigInt &x, const BigInt &y) { return mulFast(x, y); }
```
